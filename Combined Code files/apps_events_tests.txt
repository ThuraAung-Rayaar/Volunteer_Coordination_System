from datetime import timedelta
from django.utils import timezone
from rest_framework import status
from rest_framework.test import APITestCase

from apps.accounts.models import CoordinatorProfile, User
from apps.accounts.services import issue_token_for_user
from apps.common.constants import (
    APPLICATION_ACCEPTED,
    APPLICATION_PENDING,
    COORDINATOR_APPROVED,
    EVENT_ACTIVE,
    EVENT_DRAFT,
    EVENT_PUBLISHED,
    VERIFICATION_VERIFIED,
)
from apps.events.models import Event, EventApplication, EventPost
from apps.events.services import add_post, create_event, publish_event


class EventsApiTests(APITestCase):
    def setUp(self):
        # 1. Setup approved coordinator
        self.coordinator = User.objects.create(
            phone_hash="hash_coord_api",
            display_name="Coord Api",
            verification_status=VERIFICATION_VERIFIED,
        )
        CoordinatorProfile.objects.create(
            user=self.coordinator,
            status=COORDINATOR_APPROVED,
            approved_at=timezone.now(),
        )
        self.coord_token = issue_token_for_user(self.coordinator)

        # 2. Setup volunteer
        self.volunteer = User.objects.create(
            phone_hash="hash_vol_api",
            display_name="Vol Api",
            physical_ok=True,
            skills=["first_aid"],
        )
        self.vol_token = issue_token_for_user(self.volunteer)

        now = timezone.now()
        self.start_at = now + timedelta(days=5)
        self.end_at = self.start_at + timedelta(hours=4)

    def test_public_events_list_hides_draft_and_deleted(self):
        # Create a published event
        pub_event = create_event(
            self.coordinator,
            {
                "title": "Visible Published Event",
                "start_at": self.start_at,
                "end_at": self.end_at,
            },
        )
        add_post(self.coordinator, pub_event, {"title": "Post 1", "required_headcount": 5})
        publish_event(self.coordinator, pub_event)

        # Create a draft event
        draft_event = create_event(
            self.coordinator,
            {
                "title": "Hidden Draft Event",
                "start_at": self.start_at,
                "end_at": self.end_at,
            },
        )

        # Create a deleted published event
        del_event = create_event(
            self.coordinator,
            {
                "title": "Hidden Deleted Event",
                "start_at": self.start_at,
                "end_at": self.end_at,
                "deleted_at": timezone.now(),
                "deleted_by": self.coordinator,
            },
        )

        # Fetch public list (unauthenticated)
        resp = self.client.get("/api/v1/events/")
        self.assertEqual(resp.status_code, status.HTTP_200_OK)

        titles = [e["title"] for e in resp.data]
        self.assertIn("Visible Published Event", titles)
        self.assertNotIn("Hidden Draft Event", titles)
        self.assertNotIn("Hidden Deleted Event", titles)

    def test_coordinator_publish_action_endpoint(self):
        # 1. Coordinator creates draft event
        self.client.credentials(HTTP_AUTHORIZATION=f"Token {self.coord_token.key}")
        create_resp = self.client.post(
            "/api/v1/coordinator/events/",
            {
                "title": "New Coordinator Event",
                "start_at": self.start_at.isoformat(),
                "end_at": self.end_at.isoformat(),
            },
            format="json",
        )
        self.assertEqual(create_resp.status_code, status.HTTP_201_CREATED)
        event_id = create_resp.data["id"]
        self.assertEqual(create_resp.data["status"], EVENT_DRAFT)

        # 2. Attempt publish without post -> 400 DomainError (post_required)
        pub_fail = self.client.post(f"/api/v1/events/{event_id}/publish/")
        self.assertEqual(pub_fail.status_code, status.HTTP_400_BAD_REQUEST)
        self.assertEqual(pub_fail.data["error"]["code"], "post_required")

        # 3. Add post
        post_resp = self.client.post(
            f"/api/v1/events/{event_id}/posts/",
            {"title": "Helper", "required_headcount": 2},
            format="json",
        )
        self.assertEqual(post_resp.status_code, status.HTTP_201_CREATED)

        # 4. Publish event succeeds
        pub_ok = self.client.post(f"/api/v1/events/{event_id}/publish/")
        self.assertEqual(pub_ok.status_code, status.HTTP_200_OK)
        self.assertEqual(pub_ok.data["status"], EVENT_PUBLISHED)

    def test_apply_endpoint_returns_application_status(self):
        # Setup published event
        event = create_event(
            self.coordinator,
            {
                "title": "Health Checkup",
                "start_at": self.start_at,
                "end_at": self.end_at,
                "acceptance_mode": "auto",
                "max_participants": 5,
            },
        )
        post = add_post(self.coordinator, event, {"title": "Registrar", "required_headcount": 2})
        publish_event(self.coordinator, event)

        # Volunteer applies via API
        self.client.credentials(HTTP_AUTHORIZATION=f"Token {self.vol_token.key}")
        apply_resp = self.client.post(
            f"/api/v1/events/{event.id}/apply/",
            {"post_id": post.id, "source": "browse"},
            format="json",
        )
        self.assertEqual(apply_resp.status_code, status.HTTP_201_CREATED)
        self.assertEqual(apply_resp.data["status"], APPLICATION_ACCEPTED)
        self.assertEqual(apply_resp.data["post"], post.id)
        self.assertEqual(apply_resp.data["event"], event.id)

    def test_application_accept_and_roster_endpoints(self):
        event = create_event(
            self.coordinator,
            {
                "title": "Manual Event",
                "start_at": self.start_at,
                "end_at": self.end_at,
                "acceptance_mode": "manual",
            },
        )
        post = add_post(self.coordinator, event, {"title": "Guide", "required_headcount": 2})
        publish_event(self.coordinator, event)

        # Volunteer applies -> status pending
        self.client.credentials(HTTP_AUTHORIZATION=f"Token {self.vol_token.key}")
        app_resp = self.client.post(f"/api/v1/events/{event.id}/apply/", {"post_id": post.id}, format="json")
        self.assertEqual(app_resp.data["status"], APPLICATION_PENDING)
        app_id = app_resp.data["id"]

        # Coordinator accepts application
        self.client.credentials(HTTP_AUTHORIZATION=f"Token {self.coord_token.key}")
        accept_resp = self.client.post(f"/api/v1/applications/{app_id}/accept/", format="json")
        self.assertEqual(accept_resp.status_code, status.HTTP_200_OK)
        self.assertEqual(accept_resp.data["application"]["status"], APPLICATION_ACCEPTED)

        # Coordinator checks roster
        roster_resp = self.client.get(f"/api/v1/events/{event.id}/roster/")
        self.assertEqual(roster_resp.status_code, status.HTTP_200_OK)
        self.assertEqual(len(roster_resp.data["roster"]), 1)
        self.assertEqual(roster_resp.data["roster"][0]["display_name"], self.volunteer.display_name)

class EventApplicationsTests(TestCase):
    def setUp(self):
        self.coordinator = User.objects.create(
            phone_hash="coord_hash_apps",
            display_name="Coordinator Apps",
            verification_status=VERIFICATION_VERIFIED,
        )
        CoordinatorProfile.objects.create(
            user=self.coordinator,
            status=COORDINATOR_APPROVED,
            approved_at=timezone.now(),
        )

        self.volunteer1 = User.objects.create(
            phone_hash="vol_hash_1",
            display_name="Volunteer 1",
            physical_ok=True,
            can_lift_heavy=True,
            skills=["first_aid", "driving"],
        )
        self.volunteer2 = User.objects.create(
            phone_hash="vol_hash_2",
            display_name="Volunteer 2",
            physical_ok=True,
            skills=["first_aid"],
        )

        now = timezone.now()
        self.event = create_event(
            self.coordinator,
            {
                "title": "Medical Camp",
                "start_at": now + timedelta(days=7),
                "end_at": now + timedelta(days=7, hours=5),
                "application_deadline": now + timedelta(days=3),
                "acceptance_mode": "auto",
                "max_participants": 2,
            },
        )
        self.post = add_post(
            self.coordinator,
            self.event,
            {
                "title": "First Aid Responder",
                "required_headcount": 1,
                "required_skills": ["first_aid"],
            },
        )
        publish_event(self.coordinator, self.event)

    def test_duplicate_active_application_blocked(self):
        # 1. Apply to event
        app1 = apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(app1.status, APPLICATION_ACCEPTED)

        # 2. Applying again must raise duplicate application error
        with self.assertRaises(DomainError) as ctx:
            apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(ctx.exception.code, "duplicate_application")

    def test_auto_acceptance_with_capacity_creates_accepted(self):
        # Capacity is 1 on post, volunteer1 applies
        app = apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(app.status, APPLICATION_ACCEPTED)
        self.assertIsNotNone(app.decided_at)

        # Because auto_assign_enabled is True, assignment should be created automatically
        assignment = Assignment.objects.filter(application=app, user=self.volunteer1).first()
        self.assertIsNotNone(assignment)
        self.assertEqual(assignment.status, ASSIGNMENT_ASSIGNED)
        self.assertEqual(assignment.assigned_by, "auto")

    def test_auto_acceptance_full_capacity_waitlists(self):
        # Fill capacity
        app1 = apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(app1.status, APPLICATION_ACCEPTED)

        # Second volunteer applies for the same post whose capacity is 1
        app2 = apply_to_event(self.volunteer2, self.event, post=self.post)
        self.assertEqual(app2.status, APPLICATION_WAITLISTED)
        self.assertIsNotNone(app2.decided_at)

    def test_past_deadline_application_fails(self):
        # Set deadline to 1 hour in the past
        self.event.application_deadline = timezone.now() - timedelta(hours=1)
        self.event.save()

        with self.assertRaises(DomainError) as ctx:
            apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(ctx.exception.code, "deadline_passed")

    def test_manual_application_and_accept_creates_assignment(self):
        # Set event to manual acceptance mode
        self.event.acceptance_mode = "manual"
        self.event.save()

        # Apply -> status should be pending
        app = apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(app.status, APPLICATION_PENDING)
        self.assertIsNone(app.decided_at)

        # Coordinator accepts application
        accepted_app, assignment = accept_application(self.coordinator, app)
        self.assertEqual(accepted_app.status, APPLICATION_ACCEPTED)
        self.assertEqual(accepted_app.decided_by, self.coordinator)
        self.assertIsNotNone(accepted_app.decided_at)
        self.assertIsNotNone(assignment)
        self.assertEqual(assignment.assigned_by, "coordinator")
        self.assertEqual(assignment.assigned_by_user, self.coordinator)

    def test_withdraw_application_cancels_assignment(self):
        app = apply_to_event(self.volunteer1, self.event, post=self.post)
        self.assertEqual(app.status, APPLICATION_ACCEPTED)
        assignment = Assignment.objects.get(application=app)

        # Volunteer withdraws
        withdrawn_app = withdraw_application(self.volunteer1, app)
        self.assertEqual(withdrawn_app.status, APPLICATION_WITHDRAWN)

        assignment.refresh_from_db()
        self.assertEqual(assignment.status, ASSIGNMENT_CANCELLED)

class EventLifecycleTests(TestCase):
    def setUp(self):
        self.coordinator = User.objects.create(
            phone_hash="coord_hash_1",
            display_name="Coordinator One",
            verification_status=VERIFICATION_VERIFIED,
        )
        CoordinatorProfile.objects.create(
            user=self.coordinator,
            status=COORDINATOR_APPROVED,
            approved_at=timezone.now(),
        )

        self.volunteer = User.objects.create(
            phone_hash="vol_hash_1",
            display_name="Volunteer One",
        )

        now = timezone.now()
        self.start_at = now + timedelta(days=5)
        self.end_at = self.start_at + timedelta(hours=4)

    def test_publish_requires_at_least_one_post(self):
        # 1. Create draft event
        event = create_event(
            self.coordinator,
            {
                "title": "Beach Cleanup",
                "start_at": self.start_at,
                "end_at": self.end_at,
            },
        )
        self.assertEqual(event.status, EVENT_DRAFT)

        # 2. Attempt publish without posts -> fails
        with self.assertRaises(DomainError) as ctx:
            publish_event(self.coordinator, event)
        self.assertEqual(ctx.exception.code, "post_required")

        # 3. Add post and publish -> succeeds
        add_post(self.coordinator, event, {"title": "Trash Collector", "required_headcount": 10})
        published_event = publish_event(self.coordinator, event)
        self.assertEqual(published_event.status, EVENT_PUBLISHED)

    def test_invalid_lifecycle_transition_raises_error(self):
        event = create_event(
            self.coordinator,
            {
                "title": "Park Tree Planting",
                "start_at": self.start_at,
                "end_at": self.end_at,
            },
        )
        add_post(self.coordinator, event, {"title": "Planter", "required_headcount": 5})

        # Cannot directly complete or activate draft
        with self.assertRaises(InvalidStateError):
            activate_event(self.coordinator, event)

        with self.assertRaises(InvalidStateError):
            complete_event(self.coordinator, event)

        # Valid: draft -> published
        publish_event(self.coordinator, event)
        self.assertEqual(event.status, EVENT_PUBLISHED)

        # Valid: published -> active
        activate_event(self.coordinator, event)
        self.assertEqual(event.status, EVENT_ACTIVE)

        # Cannot activate already active event
        with self.assertRaises(InvalidStateError):
            activate_event(self.coordinator, event)

        # Valid: active -> completed
        complete_event(self.coordinator, event)
        self.assertEqual(event.status, EVENT_COMPLETED)

        # Cannot cancel completed event
        with self.assertRaises(InvalidStateError):
            cancel_event(self.coordinator, event)

    def test_onboarding_text_change_increments_version(self):
        event = create_event(
            self.coordinator,
            {
                "title": "Soup Kitchen",
                "start_at": self.start_at,
                "end_at": self.end_at,
                "onboarding_text": "Please bring a clean apron.",
            },
        )
        self.assertEqual(event.onboarding_version, 1)

        # Update without changing onboarding text
        update_event(self.coordinator, event, {"location_name": "Community Center"})
        event.refresh_from_db()
        self.assertEqual(event.onboarding_version, 1)

        # Update onboarding text -> increments version to 2
        update_event(self.coordinator, event, {"onboarding_text": "Please bring clean apron and closed shoes."})
        event.refresh_from_db()
        self.assertEqual(event.onboarding_version, 2)

    def test_complete_event_cascades_status_to_assignments(self):
        event = create_event(
            self.coordinator,

                "title": "Charity Marathon",

        attend.refresh_from_db()

from datetime import timedelta
from types import SimpleNamespace
from django.core.exceptions import ValidationError
from django.db.models import F, Q
from django.test import SimpleTestCase
from django.utils import timezone

from apps.accounts.models import User
from apps.common.constants import (

from apps.events.models import (

class EventModelConstraintTests(SimpleTestCase):
    def _create_dummy_user(self, user_id=1):
        return User(id=user_id, phone_hash=f"hash_{user_id}", display_name=f"User {user_id}")

    def test_event_time_constraint(self):
        coordinator = self._create_dummy_user(1)
        now = timezone.now()

        # 1. end_at earlier than start_at MUST raise ValidationError
        invalid_event = Event(
            coordinator=coordinator,
            title="Time Inverted Event",
            start_at=now + timedelta(days=2),
            end_at=now + timedelta(days=1),
        )
        with self.assertRaises(ValidationError):
            invalid_event.clean()

        # 2. end_at equal to start_at MUST raise ValidationError
        invalid_equal_event = Event(
            coordinator=coordinator,
            title="Equal Time Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=1),
        )
        with self.assertRaises(ValidationError):
            invalid_equal_event.clean()

        # 3. end_at > start_at is valid
        valid_event = Event(
            coordinator=coordinator,
            title="Valid Time Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
        )
        try:
            valid_event.clean()
        except ValidationError as e:
            self.assertNotIn("end_at", e.message_dict)

        # 4. application_deadline > start_at MUST raise ValidationError
        invalid_deadline_event = Event(
            coordinator=coordinator,
            title="Late Deadline Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
            application_deadline=now + timedelta(days=3),
        )
        with self.assertRaises(ValidationError):
            invalid_deadline_event.clean()

        # 5. Verify CheckConstraint definitions in Meta
        time_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_time")
        self.assertEqual(time_constraint.condition, Q(end_at__gt=F("start_at")))

        deadline_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_deadline")
        self.assertEqual(
            deadline_constraint.condition,
            Q(application_deadline__isnull=True) | Q(application_deadline__lte=F("start_at")),
        )

    def test_event_soft_delete_both_fields_required(self):
        coordinator = self._create_dummy_user(1)
        admin_user = self._create_dummy_user(99)
        now = timezone.now()

        # 1. deleted_at set without deleted_by MUST raise ValidationError
        orphan_deleted_at = Event(
            coordinator=coordinator,
            title="Orphan Deleted At Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
            deleted_at=now,
            deleted_by=None,
        )
        with self.assertRaises(ValidationError):
            orphan_deleted_at.clean()

        # 2. deleted_by set without deleted_at MUST raise ValidationError
        orphan_deleted_by = Event(
            coordinator=coordinator,
            title="Orphan Deleted By Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
            deleted_at=None,
            deleted_by=admin_user,
        )
        with self.assertRaises(ValidationError):
            orphan_deleted_by.clean()

        # 3. Both set is valid
        valid_soft_deleted = Event(
            coordinator=coordinator,
            title="Properly Deleted Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
            deleted_at=now,
            deleted_by=admin_user,
        )
        try:
            valid_soft_deleted.clean()
        except ValidationError as e:
            self.assertNotIn("deleted_at", str(e))
            self.assertNotIn("deleted_by", str(e))

        # 4. Both None is valid (active event)
        valid_active_event = Event(
            coordinator=coordinator,
            title="Active Not Deleted Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
            deleted_at=None,
            deleted_by=None,
        )
        try:
            valid_active_event.clean()
        except ValidationError as e:
            self.assertNotIn("deleted_at", str(e))
            self.assertNotIn("deleted_by", str(e))

        # 5. Verify CheckConstraint definition
        soft_delete_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_soft_delete")
        self.assertEqual(
            soft_delete_constraint.condition,
            (Q(deleted_at__isnull=True) & Q(deleted_by__isnull=True))
            | (Q(deleted_at__isnull=False) & Q(deleted_by__isnull=False)),
        )

    def test_event_application_unique_active_constraint(self):
        # 1. Verify UniqueConstraint on active applications
        unique_active_constraint = next(
            c for c in EventApplication._meta.constraints
            if c.name == "uq_event_application_active"
        )
        self.assertEqual(list(unique_active_constraint.fields), ["event", "user"])
        self.assertEqual(
            unique_active_constraint.condition,
            Q(status__in=["pending", "accepted", "waitlisted"]),
        )

        # 2. qr_walkin source without qr_token MUST raise ValidationError
        dummy_event = Event(id=1, title="Test Event")
        dummy_user = self._create_dummy_user(2)

        invalid_walkin = EventApplication(
            event=dummy_event,
            user=dummy_user,
            source="qr_walkin",
            qr_token=None,
        )
        with self.assertRaises(ValidationError):
            invalid_walkin.clean()

        # 3. Accepted/rejected application without decided_at MUST raise ValidationError
        invalid_accepted = EventApplication(
            event=dummy_event,
            user=dummy_user,
            source="browse",
            status=APPLICATION_ACCEPTED,
            decided_at=None,
        )
        with self.assertRaises(ValidationError):
            invalid_accepted.clean()

        invalid_rejected = EventApplication(
            event=dummy_event,
            user=dummy_user,
            source="browse",
            status=APPLICATION_REJECTED,
            decided_at=None,
        )
        with self.assertRaises(ValidationError):
            invalid_rejected.clean()

        # 4. Verify CheckConstraints in Meta
        walkin_constraint = next(
            c for c in EventApplication._meta.constraints
            if c.name == "ck_application_walkin_token"
        )
        self.assertEqual(
            walkin_constraint.condition,
            ~Q(source="qr_walkin") | Q(qr_token__isnull=False),
        )

        decided_constraint = next(
            c for c in EventApplication._meta.constraints
            if c.name == "ck_application_decided"
        )
        self.assertEqual(
            decided_constraint.condition,
            ~Q(status__in=["accepted", "rejected"]) | Q(decided_at__isnull=False),
        )
