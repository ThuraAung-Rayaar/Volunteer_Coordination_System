-- =============================================================
-- Volunteer Coordination System
-- Final Database Design v0.5
-- Stack: Django (API + admin web) + PostgreSQL + React Native client
-- Scope: online-only coordination. Server DB is the source of truth.
--
-- Conventions
--   * Integer BIGSERIAL ids, TIMESTAMPTZ everywhere.
--   * phone_hash must be an HMAC (server secret), not a plain hash.
--   * Images: only JPEG, only the server file path is stored.
--   * Cross-table consistency that Django cannot express as a FK
--     (e.g. post_id belongs to the same event_id) is enforced in the
--     service layer.
-- =============================================================

-- =============================================================
-- 2. USERS
-- =============================================================

CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,

    -- authentication (phone OTP for app; password only for admin web login)
    phone_hash TEXT NOT NULL UNIQUE,
    phone_enc  TEXT NOT NULL,
    password   TEXT,                       -- Django password hash, admin web only
    last_login TIMESTAMPTZ,

    display_name TEXT NOT NULL,

    is_admin BOOLEAN NOT NULL DEFAULT FALSE,

    -- active  : normal
    -- blocked : cannot log in / apply (blocked_until NULL = indefinite)
    -- closed  : account closed
    account_status TEXT NOT NULL DEFAULT 'active'
        CHECK (account_status IN ('active', 'blocked', 'closed')),
    blocked_until TIMESTAMPTZ,

    preferred_language TEXT NOT NULL DEFAULT 'en',
    home_area TEXT,

    -- emergency contact
    emergency_contact_name      TEXT,
    emergency_contact_phone_enc TEXT,

    -- personal details relevant to charity / volunteering
    blood_type TEXT NOT NULL DEFAULT 'unknown'
        CHECK (blood_type IN (
            'A+', 'A-', 'B+', 'B-',
            'AB+', 'AB-', 'O+', 'O-',
            'unknown'
        )),

    is_willing_to_donate        BOOLEAN NOT NULL DEFAULT FALSE,
    is_willing_to_donate_blood  BOOLEAN NOT NULL DEFAULT FALSE,
    is_willing_to_donate_items  BOOLEAN NOT NULL DEFAULT FALSE,

    -- assignment capability hints (self-declared, not medical clearance)
    physical_ok           BOOLEAN NOT NULL DEFAULT FALSE,
    good_eyesight         BOOLEAN NOT NULL DEFAULT TRUE,
    needs_glasses         BOOLEAN NOT NULL DEFAULT FALSE,
    can_lift_heavy        BOOLEAN NOT NULL DEFAULT FALSE,
    can_work_outdoor      BOOLEAN NOT NULL DEFAULT FALSE,
    has_leader_experience BOOLEAN NOT NULL DEFAULT FALSE,
    new_to_volunteer      BOOLEAN NOT NULL DEFAULT TRUE,

    -- free-form skill tags (normalized lowercase in Django)
    skills TEXT[] NOT NULL DEFAULT '{}',

    -- recruiting support preferences
    needs_meal_support   BOOLEAN NOT NULL DEFAULT FALSE,
    needs_travel_support BOOLEAN NOT NULL DEFAULT FALSE,
    accepts_unpaid       BOOLEAN NOT NULL DEFAULT TRUE,

    -- derived from user_verification_images review
    verification_status TEXT NOT NULL DEFAULT 'none'
        CHECK (verification_status IN ('none', 'pending', 'verified', 'rejected')),

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_users_blocked_until
        CHECK (blocked_until IS NULL OR account_status = 'blocked'),

    CONSTRAINT ck_users_donation_consistency
        CHECK (
            is_willing_to_donate
            OR (NOT is_willing_to_donate_blood AND NOT is_willing_to_donate_items)
        )
);

CREATE INDEX idx_users_account_status      ON users(account_status);
CREATE INDEX idx_users_verification_status ON users(verification_status);
CREATE INDEX idx_users_is_admin            ON users(is_admin) WHERE is_admin;
CREATE INDEX idx_users_blood_type          ON users(blood_type) WHERE is_willing_to_donate_blood;

CREATE TABLE user_charity_interests (

-- =============================================================
-- 4. COORDINATOR PROFILES
-- A user may create events only when status = 'approved'
-- (checked in Django).
-- =============================================================

CREATE TABLE coordinator_profiles (
    user_id BIGINT PRIMARY KEY REFERENCES users(id),

    status TEXT NOT NULL DEFAULT 'pending'
        CHECK (status IN ('pending', 'approved', 'suspended', 'revoked')),

    approved_by BIGINT REFERENCES users(id),
    approved_at TIMESTAMPTZ,

    notes TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_coordinator_approval
        CHECK (status = 'pending' OR approved_at IS NOT NULL)
);

CREATE INDEX idx_coordinator_profiles_status ON coordinator_profiles(status);


-- =============================================================
-- 5. USER VERIFICATION IMAGES
-- Backend converts/stores JPEG only. DB stores the server path.
-- Used for selfie-vs-ID photo comparison.
-- =============================================================

CREATE TABLE user_verification_images (
    id BIGSERIAL PRIMARY KEY,

    user_id BIGINT NOT NULL REFERENCES users(id),

    image_type TEXT NOT NULL
        CHECK (image_type IN ('mock_selfie', 'id_license', 'driver_license')),

    image_path  TEXT NOT NULL,
    file_format TEXT NOT NULL DEFAULT 'jpeg'
        CHECK (file_format = 'jpeg'),
    file_sha256 TEXT,                      -- integrity / duplicate detection

    -- human review result
    status TEXT NOT NULL DEFAULT 'pending'
        CHECK (status IN ('pending', 'approved', 'rejected')),

    -- automatic or manual photo comparison result
    comparison_status TEXT NOT NULL DEFAULT 'pending'
        CHECK (comparison_status IN ('pending', 'match', 'no_match', 'needs_review')),
    comparison_score NUMERIC(5,4)
        CHECK (comparison_score IS NULL OR comparison_score BETWEEN 0 AND 1),

    is_current BOOLEAN NOT NULL DEFAULT TRUE,   -- old uploads kept as history

    reviewed_by BIGINT REFERENCES users(id),
    reviewed_at TIMESTAMPTZ,
    note TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_verification_image_path
        CHECK (image_path <> '' AND lower(image_path) ~ '\.jpe?g$'),

    CONSTRAINT ck_verification_review
        CHECK (status = 'pending' OR (reviewed_by IS NOT NULL AND reviewed_at IS NOT NULL))
);

CREATE INDEX idx_user_verification_images_user   ON user_verification_images(user_id);
CREATE INDEX idx_user_verification_images_status ON user_verification_images(status)
    WHERE status = 'pending';

-- one current image per type per user
CREATE UNIQUE INDEX uq_current_verification_image
    ON user_verification_images(user_id, image_type)
    WHERE is_current;

-- =============================================================
-- 7. EVENTS
-- =============================================================

CREATE TABLE events (
    id BIGSERIAL PRIMARY KEY,

    -- must be a coordinator (approval state checked in Django)
    coordinator_id BIGINT NOT NULL REFERENCES coordinator_profiles(user_id),

    title       TEXT NOT NULL,
    description TEXT,

    purpose_type TEXT NOT NULL DEFAULT 'charity_event'
        CHECK (purpose_type IN ('charity_event', 'help_wanted_event')),

    location_name TEXT,
    landmark_text TEXT,

    start_at TIMESTAMPTZ NOT NULL,
    end_at   TIMESTAMPTZ NOT NULL,
    application_deadline TIMESTAMPTZ,

    status TEXT NOT NULL DEFAULT 'draft'
        CHECK (status IN ('draft', 'published', 'active', 'completed', 'cancelled')),

    acceptance_mode TEXT NOT NULL DEFAULT 'manual'
        CHECK (acceptance_mode IN ('auto', 'manual')),

    auto_assign_enabled BOOLEAN NOT NULL DEFAULT TRUE,

    group_size INTEGER CHECK (group_size > 0),

    leader_strategy TEXT NOT NULL DEFAULT 'none'
        CHECK (leader_strategy IN ('none', 'first_experienced', 'manual')),

    max_participants INTEGER CHECK (max_participants > 0),

    -- recruiting conditions (informational only)
    meals_provided          BOOLEAN NOT NULL DEFAULT FALSE,
    lodging_provided        BOOLEAN NOT NULL DEFAULT FALSE,
    travel_support_provided BOOLEAN NOT NULL DEFAULT FALSE,

    compensation_kind TEXT NOT NULL DEFAULT 'unpaid'
        CHECK (compensation_kind IN ('unpaid', 'arranged_outside_system')),
    compensation_note TEXT,

    -- onboarding
    onboarding_text        TEXT,
    onboarding_version     INTEGER NOT NULL DEFAULT 1 CHECK (onboarding_version >= 1),
    require_onboarding_ack BOOLEAN NOT NULL DEFAULT TRUE,

    -- walk-in QR registration
    qr_registration_enabled BOOLEAN NOT NULL DEFAULT TRUE,

    -- admin soft delete
    deleted_at TIMESTAMPTZ,
    deleted_by BIGINT REFERENCES users(id),

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_events_time
        CHECK (end_at > start_at),
    CONSTRAINT ck_events_deadline
        CHECK (application_deadline IS NULL OR application_deadline <= start_at),
    CONSTRAINT ck_events_soft_delete
        CHECK ((deleted_at IS NULL) = (deleted_by IS NULL))
);

CREATE INDEX idx_events_coordinator ON events(coordinator_id);
CREATE INDEX idx_events_start_at    ON events(start_at);

-- main listing query: visible events by status and date
CREATE INDEX idx_events_visible
    ON events(status, start_at)
    WHERE deleted_at IS NULL;

    required_headcount INTEGER NOT NULL CHECK (required_headcount > 0),

    revoked_at TIMESTAMPTZ,

        CHECK (max_uses IS NULL OR used_count <= max_uses)

-- =============================================================
-- 11. EVENT APPLICATIONS
-- =============================================================

CREATE TABLE event_applications (
    id BIGSERIAL PRIMARY KEY,

    event_id BIGINT NOT NULL REFERENCES events(id),
    post_id  BIGINT REFERENCES event_posts(id),
    user_id  BIGINT NOT NULL REFERENCES users(id),

    qr_token_id BIGINT REFERENCES qr_tokens(id),

    source TEXT NOT NULL DEFAULT 'browse'
        CHECK (source IN ('browse', 'qr_walkin', 'invite')),

    status TEXT NOT NULL DEFAULT 'pending'
        CHECK (status IN (
            'pending', 'accepted', 'waitlisted',
            'rejected', 'withdrawn', 'expired'
        )),

    applied_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    -- decided_by NULL + decided_at set = decided automatically by system
    decided_by BIGINT REFERENCES users(id),
    decided_at TIMESTAMPTZ,

    conditions_acknowledged_at TIMESTAMPTZ,
    conditions_snapshot JSONB,

    note TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_application_walkin_token
        CHECK (source <> 'qr_walkin' OR qr_token_id IS NOT NULL),
    CONSTRAINT ck_application_decided
        CHECK (status NOT IN ('accepted', 'rejected') OR decided_at IS NOT NULL)
);

CREATE INDEX idx_event_applications_event  ON event_applications(event_id);
CREATE INDEX idx_event_applications_user   ON event_applications(user_id);
CREATE INDEX idx_event_applications_status ON event_applications(event_id, status);

-- one active application per user per event
CREATE UNIQUE INDEX uq_event_application_active
    ON event_applications(event_id, user_id)
    WHERE status IN ('pending', 'accepted', 'waitlisted');


-- =============================================================
-- 12. ASSIGNMENTS
-- =============================================================

CREATE TABLE assignments (
    id BIGSERIAL PRIMARY KEY,

    event_id       BIGINT NOT NULL REFERENCES events(id),
    application_id BIGINT NOT NULL REFERENCES event_applications(id),
    post_id        BIGINT REFERENCES event_posts(id),
    user_id        BIGINT NOT NULL REFERENCES users(id),

    group_code      TEXT,
    is_group_leader BOOLEAN NOT NULL DEFAULT FALSE,

    status TEXT NOT NULL DEFAULT 'assigned'
        CHECK (status IN ('assigned', 'confirmed', 'completed', 'cancelled')),

    assigned_by         TEXT NOT NULL DEFAULT 'auto'
        CHECK (assigned_by IN ('auto', 'coordinator', 'admin')),
    assigned_by_user_id BIGINT REFERENCES users(id),

    schedule_start TIMESTAMPTZ,
    schedule_end   TIMESTAMPTZ,

    conditions_snapshot JSONB,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_assignment_actor
        CHECK ((assigned_by = 'auto') = (assigned_by_user_id IS NULL)),
    CONSTRAINT ck_assignment_schedule
        CHECK (schedule_start IS NULL OR schedule_end IS NULL OR schedule_end > schedule_start),
    CONSTRAINT ck_assignment_leader_group
        CHECK (NOT is_group_leader OR group_code IS NOT NULL)
);

CREATE INDEX idx_assignments_event ON assignments(event_id);
CREATE INDEX idx_assignments_user  ON assignments(user_id);
CREATE INDEX idx_assignments_post  ON assignments(post_id);

-- one active assignment per user per event
CREATE UNIQUE INDEX uq_assignment_active
    ON assignments(event_id, user_id)
    WHERE status <> 'cancelled';

-- one active assignment per application
CREATE UNIQUE INDEX uq_assignment_application
    ON assignments(application_id)
    WHERE status <> 'cancelled';

-- at most one leader per group
CREATE UNIQUE INDEX uq_assignment_group_leader
    ON assignments(event_id, group_code)
    WHERE is_group_leader AND status <> 'cancelled';

-- 13. ATTENDANCE (presence evidence only)

CREATE TABLE attendance (
    id BIGSERIAL PRIMARY KEY,

    event_id      BIGINT NOT NULL REFERENCES events(id),
    user_id       BIGINT NOT NULL REFERENCES users(id),
    assignment_id BIGINT REFERENCES assignments(id),
    qr_token_id   BIGINT REFERENCES qr_tokens(id),

    method TEXT
        CHECK (method IN ('qr_scan', 'coordinator_manual', 'system')),

    recorded_by_user_id BIGINT REFERENCES users(id),

    note TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_attendance_checkout
        CHECK (checkout_at IS NULL OR (checkin_at IS NOT NULL AND checkout_at >= checkin_at))

CREATE INDEX idx_attendance_event  ON attendance(event_id, status);
CREATE INDEX idx_attendance_user   ON attendance(user_id);

-- =============================================================
-- 16. ADMIN MODERATION ACTIONS
-- Admin acts as watchman and rules regulator.
-- Current state lives on users / events / feedback; this table
-- is the history of why it changed.
-- =============================================================

CREATE TABLE moderation_actions (
    id BIGSERIAL PRIMARY KEY,

    admin_user_id BIGINT NOT NULL REFERENCES users(id),

    target_user_id BIGINT REFERENCES users(id),
    event_id       BIGINT REFERENCES events(id),
    feedback_id    BIGINT REFERENCES feedback(id),

    action_type TEXT NOT NULL
        CHECK (action_type IN (
            'user_warning',
            'user_block',
            'user_unblock',
            'account_close',
            'account_reinstate',
            'event_delete',
            'event_restore',
            'event_cancel',
            'feedback_hide'
        )),

    reason TEXT,

    expires_at TIMESTAMPTZ,                -- for temporary user_block

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT ck_moderation_subject
        CHECK (
            (action_type IN ('user_warning', 'user_block', 'user_unblock',
                             'account_close', 'account_reinstate')
                AND target_user_id IS NOT NULL)
            OR (action_type IN ('event_delete', 'event_restore', 'event_cancel')
                AND event_id IS NOT NULL)
            OR (action_type = 'feedback_hide' AND feedback_id IS NOT NULL)
        ),
    CONSTRAINT ck_moderation_expiry
        CHECK (expires_at IS NULL OR action_type = 'user_block')
);

CREATE INDEX idx_moderation_actions_admin  ON moderation_actions(admin_user_id);
CREATE INDEX idx_moderation_actions_target ON moderation_actions(target_user_id, action_type);
CREATE INDEX idx_moderation_actions_event  ON moderation_actions(event_id);

-- 18. UPDATED_AT TRIGGER
-- =============================================================

CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DO $$
DECLARE
    t TEXT;
BEGIN
    FOREACH t IN ARRAY ARRAY[
        'users',
        'coordinator_profiles',
        'user_verification_images',
        'events',
        'event_posts',
        'event_applications',
        'assignments',
        'attendance'
    ]
    LOOP
        EXECUTE format(
            'CREATE TRIGGER trg_%s_updated_at
             BEFORE UPDATE ON %I
             FOR EACH ROW EXECUTE FUNCTION set_updated_at()', t, t);
    END LOOP;
END $$;

-- Remaining capacity per post
CREATE VIEW v_event_post_capacity AS
SELECT
    p.id AS post_id,
    p.event_id,
    p.title,
    p.required_headcount,
    COUNT(a.id) AS assigned_count,
    GREATEST(p.required_headcount - COUNT(a.id), 0) AS remaining
FROM event_posts p
LEFT JOIN assignments a
       ON a.post_id = p.id
      AND a.status <> 'cancelled'
GROUP BY p.id;


-- Per-event counters (subqueries avoid join fan-out)
CREATE VIEW v_event_summary AS
SELECT
    e.id AS event_id,
    e.coordinator_id,
    e.title,
    e.status,
    e.max_participants,
    e.acceptance_mode,
    e.auto_assign_enabled,
    (SELECT COUNT(*) FROM event_applications x
      WHERE x.event_id = e.id AND x.status = 'pending')    AS pending_count,
    (SELECT COUNT(*) FROM event_applications x
      WHERE x.event_id = e.id AND x.status = 'accepted')   AS accepted_count,
    (SELECT COUNT(*) FROM event_applications x
      WHERE x.event_id = e.id AND x.status = 'waitlisted') AS waitlisted_count,
    (SELECT COUNT(*) FROM assignments x
      WHERE x.event_id = e.id AND x.status <> 'cancelled') AS assignment_count,
    (SELECT COUNT(*) FROM attendance x
      WHERE x.event_id = e.id
        AND x.status IN ('checked_in', 'completed'))       AS checked_in_count
FROM events e
WHERE e.deleted_at IS NULL;


-- Accepted applications that still have no assignment
CREATE VIEW v_accepted_unassigned AS
SELECT
    app.id AS application_id,
    app.event_id,
    app.user_id,
    u.display_name,
    u.skills,
    u.physical_ok,
    u.can_lift_heavy,
    u.has_leader_experience,
    u.new_to_volunteer
FROM event_applications app
JOIN users  u ON u.id = app.user_id
JOIN events e ON e.id = app.event_id AND e.deleted_at IS NULL
WHERE app.status = 'accepted'
  AND NOT EXISTS (
      SELECT 1 FROM assignments a
       WHERE a.application_id = app.id
         AND a.status <> 'cancelled'
  );


-- Coordinator check-in board
CREATE VIEW v_event_checkin_status AS
SELECT
    a.event_id,
    u.id AS user_id,
    u.display_name,
    a.post_id,
    a.group_code,
    a.is_group_leader,
    COALESCE(att.status, 'expected') AS attendance_status,
    att.checkin_at
FROM assignments a
JOIN users u ON u.id = a.user_id
LEFT JOIN attendance att
       ON att.event_id = a.event_id
      AND att.user_id  = a.user_id
WHERE a.status <> 'cancelled';


-- Admin: warnings and blocks per user
CREATE VIEW v_user_moderation_summary AS
SELECT
    target_user_id AS user_id,
    COUNT(*) FILTER (WHERE action_type = 'user_warning') AS warning_count,
    MAX(created_at) FILTER (WHERE action_type = 'user_warning') AS last_warning_at,
    COUNT(*) FILTER (WHERE action_type = 'user_block')   AS block_count,
    MAX(created_at) FILTER (WHERE action_type = 'user_block')   AS last_block_at
FROM moderation_actions
WHERE target_user_id IS NOT NULL
GROUP BY target_user_id;


-- Admin dashboard (single row)
CREATE VIEW v_admin_dashboard AS
SELECT
    (SELECT COUNT(*) FROM users WHERE account_status = 'active')  AS active_users,
    (SELECT COUNT(*) FROM users WHERE account_status = 'blocked') AS blocked_users,
    (SELECT COUNT(*) FROM users WHERE verification_status = 'pending')
                                                                  AS users_pending_verification,
    (SELECT COUNT(*) FROM user_verification_images
      WHERE status = 'pending' AND is_current)                    AS images_pending_review,
    (SELECT COUNT(*) FROM coordinator_profiles WHERE status = 'pending')
                                                                  AS coordinators_pending_approval,
    (SELECT COUNT(*) FROM coordinator_profiles WHERE status = 'approved')
                                                                  AS coordinators_approved,
    (SELECT COUNT(*) FROM events WHERE deleted_at IS NULL AND status = 'draft')     AS events_draft,
    (SELECT COUNT(*) FROM events WHERE deleted_at IS NULL AND status = 'published') AS events_published,
    (SELECT COUNT(*) FROM events WHERE deleted_at IS NULL AND status = 'active')    AS events_active,
    (SELECT COUNT(*) FROM events WHERE deleted_at IS NULL AND status = 'completed') AS events_completed,
    (SELECT COUNT(*) FROM events WHERE deleted_at IS NULL AND status = 'cancelled') AS events_cancelled,
    (SELECT COUNT(*) FROM events WHERE deleted_at IS NOT NULL)    AS events_deleted,
    (SELECT COUNT(*) FROM moderation_actions
      WHERE action_type = 'user_warning'
        AND created_at > now() - INTERVAL '30 days')              AS warnings_last_30d;
