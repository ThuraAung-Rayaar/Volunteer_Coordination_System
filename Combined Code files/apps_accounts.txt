class UserVerificationImageAdmin(admin.ModelAdmin):
    list_display = (
        "id",
        "user",
        "image_type",
        "status",
        "comparison_status",
        "comparison_score",
        "is_current",
        "reviewed_by",
        "inspect_image",
        "created_at",
    )
    list_filter = ("status", "comparison_status", "image_type", "is_current")
    search_fields = ("user__display_name", "note")
    readonly_fields = ("image_path", "file_sha256", "inspect_image", "reviewed_by", "reviewed_at", "created_at")
    actions = ["approve_selected_images", "reject_selected_images"]

    def inspect_image(self, obj):
        if obj.image_path:
            return format_html(
                '<a href="/admin/accounts/userverificationimage/{}/view/" target="_blank" rel="noopener noreferrer">View Image</a>',
                obj.id,
            )
        return "No Image"

    def get_urls(self):
        from django.urls import path
        urls = super().get_urls()
        custom_urls = [
            path(
                "<int:image_id>/view/",
                self.admin_site.admin_view(self.secure_view_image),
                name="userverificationimage-secure-view",
            ),
        ]
        return custom_urls + urls

    def secure_view_image(self, request, image_id):
        from django.http import Http404, FileResponse
        from apps.common.permissions import ensure_admin
        ensure_admin(request.user)
        img = UserVerificationImage.objects.filter(id=image_id).first()
        if not img or not img.image_path:
            raise Http404("Image not found")
        return FileResponse(img.image_path.open("rb"), content_type="image/jpeg")

    @admin.action(description="Approve selected verification images")
    def approve_selected_images(self, request, queryset):
        for img in queryset:
            try:
                review_verification_image(
                    actor=request.user,
                    image=img,
                    status="approved",
                    note="Approved via admin action",
                )
            except Exception as e:
                self.message_user(request, f"Error approving image #{img.id}: {e}", level="error")

    @admin.action(description="Reject selected verification images")
    def reject_selected_images(self, request, queryset):
        for img in queryset:
            try:
                review_verification_image(
                    actor=request.user,
                    image=img,
                    status="rejected",
                    note="Rejected via admin action",
                )
            except Exception as e:
                self.message_user(request, f"Error rejecting image #{img.id}: {e}", level="error")

class UserManager(BaseUserManager):
    def create_user(self, phone_hash, display_name, password=None, **extra_fields):
        if not phone_hash:
            raise ValueError("The phone_hash must be set.")
        if not display_name:
            raise ValueError("The display_name must be set.")

        extra_fields.setdefault("is_staff", False)
        extra_fields.setdefault("is_superuser", False)

        user = self.model(
            phone_hash=phone_hash,
            display_name=display_name,
            **extra_fields,
        )
        if password:
            user.set_password(password)
        else:
            user.set_unusable_password()
        user.save(using=self._db)
        return user

    def create_superuser(self, phone_hash, display_name, password=None, **extra_fields):
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)
        extra_fields.setdefault("is_admin", True)

        if not password:
            raise ValueError("Superusers must have a password.")

        return self.create_user(phone_hash, display_name, password, **extra_fields)


class User(AbstractBaseUser, PermissionsMixin):
    ACCOUNT_STATUS_CHOICES = [
        (ACCOUNT_ACTIVE, "Active"),
        (ACCOUNT_BLOCKED, "Blocked"),
        (ACCOUNT_CLOSED, "Closed"),
    ]

    BLOOD_TYPE_CHOICES = [
        ("A+", "A+"),
        ("A-", "A-"),
        ("B+", "B+"),
        ("B-", "B-"),
        ("AB+", "AB+"),
        ("AB-", "AB-"),
        ("O+", "O+"),
        ("O-", "O-"),
        ("unknown", "Unknown"),
    ]

    VERIFICATION_STATUS_CHOICES = [
        (VERIFICATION_NONE, "None"),
        (VERIFICATION_PENDING, "Pending"),
        (VERIFICATION_VERIFIED, "Verified"),
        (VERIFICATION_REJECTED, "Rejected"),
    ]

    id = models.BigAutoField(primary_key=True)

    # Auth
    phone_hash = models.CharField(max_length=128, unique=True, db_index=True)
    phone_enc = models.TextField(help_text="Encrypted phone representation")
    display_name = models.CharField(max_length=150)
    password = models.CharField(max_length=128, null=True, blank=True)

    is_admin = models.BooleanField(default=False)
    is_staff = models.BooleanField(default=False)
    is_active = models.BooleanField(default=True)

    account_status = models.CharField(
        max_length=20,
        choices=ACCOUNT_STATUS_CHOICES,
        default=ACCOUNT_ACTIVE,
    )
    blocked_until = models.DateTimeField(null=True, blank=True)

    preferred_language = models.CharField(max_length=10, default="en")
    home_area = models.CharField(max_length=150, null=True, blank=True)

    # Emergency Contact
    emergency_contact_name = models.CharField(max_length=150, null=True, blank=True)
    emergency_contact_phone_enc = models.TextField(null=True, blank=True)

    # Charity / Volunteering attributes
    blood_type = models.CharField(
        max_length=10,
        choices=BLOOD_TYPE_CHOICES,
        default="unknown",
    )
    is_willing_to_donate = models.BooleanField(default=False)
    is_willing_to_donate_blood = models.BooleanField(default=False)
    is_willing_to_donate_items = models.BooleanField(default=False)

    # Physical / Capability hints
    physical_ok = models.BooleanField(default=False)
    good_eyesight = models.BooleanField(default=True)
    needs_glasses = models.BooleanField(default=False)
    can_lift_heavy = models.BooleanField(default=False)
    can_work_outdoor = models.BooleanField(default=False)
    has_leader_experience = models.BooleanField(default=False)
    new_to_volunteer = models.BooleanField(default=True)

    # Free-form skills
    skills = ChoiceArrayField(
        models.CharField(max_length=50),
        default=list,
        blank=True,
    )

    # Recruiting preferences
    needs_meal_support = models.BooleanField(default=False)
    needs_travel_support = models.BooleanField(default=False)
    accepts_unpaid = models.BooleanField(default=True)

    # Identity verification
    verification_status = models.CharField(
        max_length=20,
        choices=VERIFICATION_STATUS_CHOICES,
        default=VERIFICATION_NONE,
    )

    last_login = models.DateTimeField(null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    objects = UserManager()

    USERNAME_FIELD = "phone_hash"
    REQUIRED_FIELDS = ["display_name"]

    class Meta:
        db_table = "users"
        verbose_name = "User"
        verbose_name_plural = "Users"
        constraints = [
            models.CheckConstraint(
                condition=Q(blocked_until__isnull=True) | Q(account_status="blocked"),
                name="ck_users_blocked_until",
            ),
            models.CheckConstraint(
                condition=Q(is_willing_to_donate=True)
                | (
                    Q(is_willing_to_donate_blood=False)
                    & Q(is_willing_to_donate_items=False)
                ),
                name="ck_users_donation_consistency",
            ),
        ]
        indexes = [
            models.Index(fields=["account_status"], name="idx_users_account_status"),
            models.Index(fields=["verification_status"], name="idx_users_verif_status"),
            models.Index(
                fields=["is_admin"],
                name="idx_users_is_admin",
                condition=Q(is_admin=True),
            ),
            models.Index(
                fields=["blood_type"],
                name="idx_users_blood_donors",
                condition=Q(is_willing_to_donate_blood=True),
            ),
            GinIndex(fields=["skills"], name="idx_users_skills_gin"),
        ]

    def __str__(self):
        return f"{self.display_name} (ID: {self.id})"

    def clean(self):
        super().clean()
        if self.blocked_until is not None and self.account_status != ACCOUNT_BLOCKED:
            raise ValidationError(
                {"blocked_until": "blocked_until may only be set when account_status is 'blocked'."}
            )
        if (self.is_willing_to_donate_blood or self.is_willing_to_donate_items) and not self.is_willing_to_donate:
            raise ValidationError(
                {"is_willing_to_donate": "is_willing_to_donate must be True if willing to donate blood or items."}
            )

        (COORDINATOR_REVOKED, "Revoked"),

        on_delete=models.CASCADE,
        primary_key=True,
        related_name="coordinator_profile",
    )
    status = models.CharField(
        max_length=20,
        choices=STATUS_CHOICES,
        default=COORDINATOR_PENDING,
    )
    approved_by = models.ForeignKey(
        User,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="approved_coordinators",
    )
    approved_at = models.DateTimeField(null=True, blank=True)
    notes = models.TextField(null=True, blank=True)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = "coordinator_profiles"
        verbose_name = "Coordinator Profile"
        verbose_name_plural = "Coordinator Profiles"
        constraints = [
            models.CheckConstraint(
                condition=Q(status="pending") | Q(approved_at__isnull=False),
                name="ck_coordinator_approval",
            ),
        ]

    def __str__(self):
        return f"Coordinator: {self.user.display_name} [{self.status}]"

    def clean(self):
        super().clean()
        if self.status != COORDINATOR_PENDING and self.approved_at is None:
            raise ValidationError(
                {"approved_at": "approved_at must be set when status is not 'pending'."}
            )

class UserVerificationImage(models.Model):
    IMAGE_TYPE_CHOICES = [

        (IMAGE_ID_LICENSE, "ID License"),
        (IMAGE_DRIVER_LICENSE, "Driver License"),
    ]

    STATUS_CHOICES = [
        (VERIFICATION_PENDING, "Pending"),
        ("approved", "Approved"),
        (VERIFICATION_VERIFIED, "Verified"),
        (VERIFICATION_REJECTED, "Rejected"),
    ]

    COMPARISON_STATUS_CHOICES = [
        ("pending", "Pending"),
        ("match", "Match"),
        ("no_match", "No Match"),
        ("needs_review", "Needs Review"),
    ]

    id = models.BigAutoField(primary_key=True)
    user = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name="verification_images",
    )
    image_type = models.CharField(max_length=30, choices=IMAGE_TYPE_CHOICES)
    image_path = models.ImageField(upload_to="verification_images/")
    file_format = models.CharField(max_length=10, default="jpeg")
    file_sha256 = models.CharField(max_length=64, null=True, blank=True)

    comparison_score = models.DecimalField(
        max_digits=5,
        decimal_places=4,
        null=True,
        blank=True,
    )
    status = models.CharField(
        max_length=20,
        choices=STATUS_CHOICES,
        default=VERIFICATION_PENDING,
    )
    comparison_status = models.CharField(
        max_length=20,
        choices=COMPARISON_STATUS_CHOICES,
        default="pending",
    )
    is_current = models.BooleanField(default=True)

    reviewed_by = models.ForeignKey(
        User,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="reviewed_verification_images",
    )
    reviewed_at = models.DateTimeField(null=True, blank=True)
    note = models.TextField(null=True, blank=True)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = "user_verification_images"
        verbose_name = "User Verification Image"
        verbose_name_plural = "User Verification Images"
        constraints = [
            models.UniqueConstraint(
                fields=["user", "image_type"],
                condition=Q(is_current=True),
                name="uq_current_verification_image",
            ),
            models.CheckConstraint(
                condition=Q(status="pending")
                | (Q(reviewed_by__isnull=False) & Q(reviewed_at__isnull=False)),
                name="ck_verification_review",
            ),
            models.CheckConstraint(
                condition=Q(comparison_score__isnull=True)
                | (Q(comparison_score__gte=0) & Q(comparison_score__lte=1)),
                name="ck_verification_score_range",
            ),
        ]

    def __str__(self):
        return f"{self.user.display_name} - {self.image_type} ({self.status})"

    def clean(self):
        super().clean()
        if self.status != VERIFICATION_PENDING:
            if self.reviewed_by is None or self.reviewed_at is None:
                raise ValidationError(
                    "Reviewed images must have reviewed_by and reviewed_at set."
                )
        if self.comparison_score is not None:
            if not (0 <= self.comparison_score <= 1):
                raise ValidationError(
                    {"comparison_score": "comparison_score must be between 0 and 1."}
                )

def request_otp(phone: str, purpose: str = "login") -> dict:
    raw_phone = normalize_phone(phone)
    phone_hash = compute_phone_hash(raw_phone)

    # Check if existing account is closed
    existing_user = User.objects.filter(phone_hash=phone_hash).first()
    if existing_user and existing_user.account_status == ACCOUNT_CLOSED:
        raise AccountClosedError()

    # Generate OTP code (or fixed dev code if configured)
    if settings.DEBUG and getattr(settings, "DEV_FIXED_OTP", None):
        code = str(settings.DEV_FIXED_OTP)
    else:
        code = generate_otp_code()

    code_hash = hash_otp(code)
    expires_at = timezone.now() + timedelta(seconds=OTP_TTL_SECONDS)

    AuthOtp.objects.create(
        phone_hash=phone_hash,
        code_hash=code_hash,
        purpose=purpose,
        expires_at=expires_at,
    )

    response = {"status": "sent"}
    # In DEBUG or test environments, expose code for easy testing/local dev
    if settings.DEBUG or getattr(settings, "TESTING", False):
        response["dev_code"] = code

    return response


def verify_otp(phone: str, code: str, purpose: str = "login") -> tuple[User, Token, bool]:
    raw_phone = normalize_phone(phone)
    phone_hash = compute_phone_hash(raw_phone)

    otp = (
        AuthOtp.objects.filter(
            phone_hash=phone_hash,
            purpose=purpose,
            consumed_at__isnull=True,
        )
        .order_by("-created_at")
        .first()
    )

    if not otp or otp.expires_at <= timezone.now():
        raise InvalidOtpError("Invalid or expired OTP.")

    if otp.attempt_count >= OTP_MAX_ATTEMPTS:
        raise TooManyAttemptsError("Too many attempts. Request a new OTP.")

    # Increment attempt count and save immediately
    otp.attempt_count += 1
    otp.save(update_fields=["attempt_count"])

    # Validate OTP hash
    is_valid = hash_otp(code) == otp.code_hash
    if not is_valid and settings.DEBUG and getattr(settings, "DEV_FIXED_OTP", None):
        is_valid = str(code) == str(settings.DEV_FIXED_OTP)

    if not is_valid:
        raise InvalidOtpError("Invalid OTP.")

    with transaction.atomic():
        # Mark consumed
        otp.consumed_at = timezone.now()
        otp.save(update_fields=["consumed_at"])

        # Get or create User
        created = False
        user = User.objects.filter(phone_hash=phone_hash).first()
        if not user:
            display_name = f"Volunteer {raw_phone[-4:]}" if len(raw_phone) >= 4 else "Volunteer"
            user = User.objects.create(
                phone_hash=phone_hash,
                phone_enc=encrypt_phone(raw_phone),
                display_name=display_name,
            )
            created = True

        # Validate account status (raises AccountClosedError or AccountBlockedError)
        ensure_active_account(user)

        # Issue or retrieve token
        token, _ = Token.objects.get_or_create(user=user)

        # Log login action
        log_action(actor=user, action="login", entity_type="user", entity_id=user.id)

    return (user, token, created)


def issue_token_for_user(user: User) -> Token:
    ensure_active_account(user)
    token, _ = Token.objects.get_or_create(user=user)
    return token


def update_profile(user: User, data: dict) -> User:
    """
    Update volunteer profile.
    Disallows updating is_admin, account_status, verification_status directly.
    """
    forbidden_fields = {"is_admin", "account_status", "verification_status", "blocked_until", "id", "phone_hash", "phone_enc"}
    attempted_forbidden = forbidden_fields.intersection(data.keys())
    if attempted_forbidden:
        raise DomainError(
            f"Cannot update restricted field(s): {', '.join(sorted(attempted_forbidden))}",
            code="field_not_allowed",
        )

    clean_data = dict(data)
    with transaction.atomic():
        if "emergency_contact_phone" in clean_data:
            raw_contact_phone = clean_data.pop("emergency_contact_phone")
            if raw_contact_phone:
                clean_data["emergency_contact_phone_enc"] = encrypt_phone(normalize_phone(raw_contact_phone))
            else:
                clean_data["emergency_contact_phone_enc"] = None

        for field, value in clean_data.items():
            if hasattr(user, field):
                setattr(user, field, value)

        user.clean()
        user.save()

        log_action(
            actor=user,
            action="profile_update",
            entity_type="user",
            entity_id=user.id,
            after_data={k: v for k, v in clean_data.items() if not k.endswith("_enc")},
        )

    return user


def set_charity_interests(user: User, codes: list[str]) -> None:
    """
    Atomically replace UserCharityInterest records for user.
    """
    with transaction.atomic():
        valid_fields = list(CharityField.objects.filter(code__in=codes, active=True))
        UserCharityInterest.objects.filter(user=user).delete()
        UserCharityInterest.objects.bulk_create(
            [UserCharityInterest(user=user, field_code=field) for field in valid_fields]
        )
        log_action(
            actor=user,
            action="set_charity_interests",
            entity_type="user",
            entity_id=user.id,
            after_data={"codes": codes},
        )


def add_verification_image(user: User, image_type: str, uploaded_file) -> UserVerificationImage:
    """
    Validate and transcode uploaded image to JPEG.
    Deactivates previous images of the same image_type.
    Creates new UserVerificationImage and recomputes user verification status.
    """
    valid_types = {IMAGE_MOCK_SELFIE, IMAGE_ID_LICENSE, IMAGE_DRIVER_LICENSE}
    if image_type not in valid_types:
        raise DomainError(f"Invalid image type: {image_type}", code="invalid_image_type")

    dest_dir = Path(settings.MEDIA_ROOT) / "verification_images"
    rel_path, sha256_hex = validate_and_transcode_jpeg(uploaded_file, dest_dir)

    with transaction.atomic():
        # Deactivate old current images of the same image_type
        UserVerificationImage.objects.filter(
            user=user,
            image_type=image_type,
            is_current=True,
        ).update(is_current=False)

        new_image = UserVerificationImage.objects.create(
            user=user,
            image_type=image_type,
            image_path=rel_path,
            file_format="jpeg",
            file_sha256=sha256_hex,
            status=VERIFICATION_PENDING,
            is_current=True,
        )

        recompute_verification_status(user)

        log_action(
            actor=user,
            action="upload_verification_image",
            entity_type="verification_image",
            entity_id=new_image.id,
            after_data={"image_type": image_type, "file_sha256": sha256_hex},
        )

    return new_image


def review_verification_image(
    actor: User,
    image: UserVerificationImage,
    status: str,
    note: str = "",
) -> UserVerificationImage:
    """
    Admin review of user verification image.
    Updates status, reviewer, and recomputes verification status of target user.
    """
    ensure_admin(actor)

    valid_statuses = {"approved", "verified", VERIFICATION_VERIFIED, VERIFICATION_REJECTED}
    if status not in valid_statuses:
        raise DomainError(f"Invalid review status: {status}", code="invalid_status")

    db_status = "approved" if status in ("approved", "verified", VERIFICATION_VERIFIED) else VERIFICATION_REJECTED

    with transaction.atomic():
        image.status = db_status
        image.reviewed_by = actor
        image.reviewed_at = timezone.now()
        image.note = note
        image.save(update_fields=["status", "reviewed_by", "reviewed_at", "note"])

        recompute_verification_status(image.user)

        log_action(
            actor=actor,
            action=f"review_image_{db_status}",
            entity_type="verification_image",
            entity_id=image.id,
            after_data={"status": db_status, "note": note},
        )

    return image


def recompute_verification_status(user: User) -> None:
    """
    Recompute user's verification_status based on current verification images.
    - If any current image rejected -> rejected
    - Else if approved mock_selfie AND (approved id_license OR approved driver_license) -> verified
    - Else if any current pending -> pending
    - Else -> none
    """
    current_images = list(UserVerificationImage.objects.filter(user=user, is_current=True))

    has_rejected = any(img.status in ("rejected", VERIFICATION_REJECTED) for img in current_images)
    has_pending = any(img.status in ("pending", VERIFICATION_PENDING) for img in current_images)
    has_mock = any(
        img.status in ("approved", "verified", VERIFICATION_VERIFIED) and img.image_type == IMAGE_MOCK_SELFIE
        for img in current_images
    )
    has_license = any(
        img.status in ("approved", "verified", VERIFICATION_VERIFIED)
        and img.image_type in (IMAGE_ID_LICENSE, IMAGE_DRIVER_LICENSE)
        for img in current_images
    )

    if has_rejected:
        new_status = VERIFICATION_REJECTED
    elif has_mock and has_license:
        new_status = VERIFICATION_VERIFIED
    elif has_pending:
        new_status = VERIFICATION_PENDING
    else:
        new_status = VERIFICATION_NONE

    user.verification_status = new_status
    user.save(update_fields=["verification_status"])


def apply_as_coordinator(user: User, notes: str = "") -> CoordinatorProfile:
    """
    Active user applies to become a coordinator.
    """
    ensure_active_account(user)

    with transaction.atomic():
        profile, created = CoordinatorProfile.objects.get_or_create(
            user=user,
            defaults={"status": COORDINATOR_PENDING, "notes": notes},
        )
        if not created and profile.status in (COORDINATOR_REVOKED, COORDINATOR_SUSPENDED):
            profile.status = COORDINATOR_PENDING
            if notes:
                profile.notes = notes
            profile.save(update_fields=["status", "notes"])

        log_action(
            actor=user,
            action="apply_coordinator",
            entity_type="coordinator_profile",
            entity_id=user.id,
        )

    return profile


def approve_coordinator(actor: User, profile: CoordinatorProfile) -> CoordinatorProfile:
    """
    Admin approves coordinator application.
    Requires user.verification_status == 'verified'.
    """
    ensure_admin(actor)

    profile.user.refresh_from_db()
    if profile.user.verification_status != VERIFICATION_VERIFIED:
        raise VerificationRequiredError(
            "User must be fully verified before coordinator status can be approved."
        )

    with transaction.atomic():
        profile.status = COORDINATOR_APPROVED
        profile.approved_by = actor
        profile.approved_at = timezone.now()
        profile.save(update_fields=["status", "approved_by", "approved_at"])

        log_action(
            actor=actor,
            action="approve_coordinator",
            entity_type="coordinator_profile",
            entity_id=profile.user_id,
        )

    return profile


def suspend_coordinator(actor: User, profile: CoordinatorProfile) -> CoordinatorProfile:
    """
    Admin suspends an approved coordinator.
    """
    ensure_admin(actor)

    with transaction.atomic():
        profile.status = COORDINATOR_SUSPENDED
        profile.save(update_fields=["status"])

        log_action(
            actor=actor,
            action="suspend_coordinator",
            entity_type="coordinator_profile",
            entity_id=profile.user_id,
        )

    return profile


def revoke_coordinator(actor: User, profile: CoordinatorProfile) -> CoordinatorProfile:
    """
    Admin revokes coordinator role permanently.
    """
    ensure_admin(actor)

    with transaction.atomic():
        profile.status = COORDINATOR_REVOKED
        profile.save(update_fields=["status"])

        log_action(
            actor=actor,
            action="revoke_coordinator",
            entity_type="coordinator_profile",
            entity_id=profile.user_id,
        )

    return profile

class OtpRequestView(APIView):
    permission_classes = [AllowAny]

    def post(self, request):
        serializer = OtpRequestSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        res = request_otp(
            phone=serializer.validated_data["phone"],
            purpose=serializer.validated_data.get("purpose", "login"),
        )
        return Response(res, status=status.HTTP_200_OK)


class OtpVerifyView(APIView):
    permission_classes = [AllowAny]

    def post(self, request):
        serializer = OtpVerifySerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        user, token, created = verify_otp(
            phone=serializer.validated_data["phone"],
            code=serializer.validated_data["code"],
            purpose=serializer.validated_data.get("purpose", "login"),
        )
        return Response(
            {
                "token": token.key,
                "user": UserProfileSerializer(user).data,
                "created": created,
            },
            status=status.HTTP_200_OK,
        )


class MeProfileView(APIView):
    permission_classes = [IsAuthenticated, IsActiveAccount]

    def get(self, request):
        serializer = UserProfileSerializer(request.user)
        return Response(serializer.data, status=status.HTTP_200_OK)

    def patch(self, request):
        # We pass request.data directly to update_profile so domain service detects forbidden field tampering
        data = request.data.dict() if hasattr(request.data, "dict") else dict(request.data)
        updated_user = update_profile(request.user, data)
        return Response(UserProfileSerializer(updated_user).data, status=status.HTTP_200_OK)


class MeCharityInterestsView(APIView):
    permission_classes = [IsAuthenticated, IsActiveAccount]

    def get(self, request):
        interests = list(request.user.charity_interests.values_list("field_code_id", flat=True))
        return Response({"interests": interests}, status=status.HTTP_200_OK)

        serializer.is_valid(raise_exception=True)
        codes = serializer.validated_data["codes"]
        set_charity_interests(request.user, codes)
        return Response({"status": "ok", "interests": codes}, status=status.HTTP_200_OK)

class MeCoordinatorView(APIView):

        profile = CoordinatorProfile.objects.filter(user=request.user).first()
