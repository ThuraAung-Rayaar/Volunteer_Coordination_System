# apps/audit/tests/test_api.py
from datetime import timedelta
from django.utils import timezone
from rest_framework import status
from rest_framework.test import APITestCase

from apps.accounts.models import CoordinatorProfile, User
from apps.audit.models import ModerationAction
from apps.common.constants import COORDINATOR_APPROVED, VERIFICATION_VERIFIED
from apps.events.services import add_post, create_event, publish_event


class AuditApiTests(APITestCase):
    def setUp(self):
        # 1. Admin
        self.admin = User.objects.create(
            phone_hash="hash_admin_api",
            display_name="API Admin",
            is_admin=True,
        )

        # 2. Regular user
        self.user = User.objects.create(
            phone_hash="hash_user_api",
            display_name="API Volunteer",
            account_status="active",
        )

        # 3. Coordinator & Event
        self.coordinator = User.objects.create(
            phone_hash="hash_coord_api_mod",
            display_name="API Coordinator",
            verification_status=VERIFICATION_VERIFIED,
        )
        CoordinatorProfile.objects.create(
            user=self.coordinator,
            status=COORDINATOR_APPROVED,
            approved_at=timezone.now(),
        )

        now = timezone.now()
        self.event = create_event(
            self.coordinator,
            {
                "title": "Riverside Tree Planting",
                "start_at": now + timedelta(days=2),
                "end_at": now + timedelta(days=2, hours=4),
            },
        )
        self.post = add_post(
            self.coordinator,
            self.event,
            {"title": "Planter", "required_headcount": 5},
        )
        publish_event(self.coordinator, self.event)

    def test_dashboard_stats_endpoint_admin_only(self):
        # 1. Anonymous forbidden / unauthorized
        resp_anon = self.client.get("/api/v1/admin/dashboard/")
        self.assertIn(resp_anon.status_code, [status.HTTP_401_UNAUTHORIZED, status.HTTP_403_FORBIDDEN])

        # 2. Regular user forbidden
        self.client.force_authenticate(user=self.user)
        resp_user = self.client.get("/api/v1/admin/dashboard/")
        self.assertEqual(resp_user.status_code, status.HTTP_403_FORBIDDEN)

        # 3. Admin allowed
        self.client.force_authenticate(user=self.admin)
        resp_admin = self.client.get("/api/v1/admin/dashboard/")
        self.assertEqual(resp_admin.status_code, status.HTTP_200_OK)
        self.assertIn("active_users", resp_admin.data)
        self.assertIn("events_published", resp_admin.data)

    def test_block_user_endpoint_executes_service(self):
        self.client.force_authenticate(user=self.admin)

        resp = self.client.post(
            f"/api/v1/admin/users/{self.user.id}/block/",
            {"reason": "Terms violation."},
            format="json",
        )
        self.assertEqual(resp.status_code, status.HTTP_200_OK)
        self.assertEqual(resp.data["action_type"], "user_block")

        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "blocked")

        # Unblock user
        resp_unblock = self.client.post(
            f"/api/v1/admin/users/{self.user.id}/unblock/",
            {"reason": "Appeal accepted."},
            format="json",
        )
        self.assertEqual(resp_unblock.status_code, status.HTTP_200_OK)
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "active")

    def test_event_delete_and_restore_api(self):
        self.client.force_authenticate(user=self.admin)

        # Delete event
        del_resp = self.client.post(
            f"/api/v1/admin/events/{self.event.id}/delete/",
            {"reason": "Content violation."},
            format="json",
        )
        self.assertEqual(del_resp.status_code, status.HTTP_200_OK)
        self.event.refresh_from_db()
        self.assertIsNotNone(self.event.deleted_at)

        # Restore event
        res_resp = self.client.post(
            f"/api/v1/admin/events/{self.event.id}/restore/",
            {"reason": "Resolved."},
            format="json",
        )
        self.assertEqual(res_resp.status_code, status.HTTP_200_OK)
        self.event.refresh_from_db()
        self.assertIsNone(self.event.deleted_at)

    def test_user_warn_close_reinstate_api(self):
        self.client.force_authenticate(user=self.admin)

        # 1. Warn user
        warn_resp = self.client.post(
            f"/api/v1/admin/users/{self.user.id}/warn/",
            {"reason": "Inappropriate comment warning."},
            format="json",
        )
        self.assertEqual(warn_resp.status_code, status.HTTP_200_OK)
        self.assertEqual(warn_resp.data["action_type"], "user_warning")

        # 2. Close account
        close_resp = self.client.post(
            f"/api/v1/admin/users/{self.user.id}/close/",
            {"reason": "Severe policy violation."},
            format="json",
        )
        self.assertEqual(close_resp.status_code, status.HTTP_200_OK)
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "closed")

        # 3. Reinstate account
        rein_resp = self.client.post(
            f"/api/v1/admin/users/{self.user.id}/reinstate/",
            {"reason": "Account reinstated after review."},
            format="json",
        )
        self.assertEqual(rein_resp.status_code, status.HTTP_200_OK)
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "active")

    def test_event_cancel_and_feedback_hide_api(self):
        from apps.events.services import activate_event, complete_event
        from apps.ops.models import Feedback
        from apps.ops.services import create_assignment_from_application, submit_feedback
        from apps.events.models import EventApplication

        # Create participating assignment and completed event for feedback
        app = EventApplication.objects.create(
            event=self.event,
            post=self.post,
            user=self.user,
            status="accepted",
            decided_at=timezone.now(),
        )
        create_assignment_from_application(app)
        activate_event(self.coordinator, self.event)
        complete_event(self.coordinator, self.event)

        fb = submit_feedback(
            user=self.user,
            event=self.event,
            direction="volunteer_to_event",
            score=1,
            comment="Bad language content.",
        )

        self.client.force_authenticate(user=self.admin)

        # Hide feedback
        hide_resp = self.client.post(
            f"/api/v1/admin/feedback/{fb.id}/hide/",
            {"reason": "Profanity detected."},
            format="json",
        )
        self.assertEqual(hide_resp.status_code, status.HTTP_200_OK)
        fb.refresh_from_db()
        self.assertEqual(fb.status, "hidden")

        # Create another event to test cancel
        now = timezone.now()
        event2 = create_event(

        add_post(self.coordinator, event2, {"title": "Relief Post", "required_headcount": 5})
        publish_event(self.coordinator, event2)

        cancel_resp = self.client.post(
            f"/api/v1/admin/events/{event2.id}/cancel/",
            {"reason": "Cancelled due to river overflow risk."},

        event2.refresh_from_db()

# apps/audit/tests/test_models.py
from datetime import timedelta
from django.core.exceptions import ValidationError
from django.db.models import Q
from django.test import SimpleTestCase
from django.utils import timezone

from apps.accounts.models import User
from apps.audit.models import AuditLog, ModerationAction
from apps.events.models import Event
from apps.ops.models import Feedback


class AuditModelConstraintTests(SimpleTestCase):
    def _create_dummy_user(self, user_id=1):
        return User(id=user_id, phone_hash=f"hash_{user_id}", display_name=f"User {user_id}")

    def test_audit_log_append_only(self):
        actor = self._create_dummy_user(1)
        log = AuditLog(
            id=10,
            actor_user=actor,
            action="user_created",
            entity_type="user",
            entity_id=1,
        )
        # Mark as already persisted
        log._state.adding = False

        # 1. Updating existing AuditLog MUST raise ValidationError
        with self.assertRaises(ValidationError) as ctx:
            log.save()
        self.assertIn("append-only", str(ctx.exception))

        # 2. Deleting AuditLog MUST raise ValidationError
        with self.assertRaises(ValidationError) as ctx:
            log.delete()
        self.assertIn("append-only", str(ctx.exception))

    def test_moderation_action_subject_validation(self):
        admin_user = self._create_dummy_user(99)
        target_user = self._create_dummy_user(10)
        event = Event(id=1, title="Test Event")
        feedback = Feedback(id=1, event=event, rater_user=target_user)

        # 1. User action without target_user MUST raise ValidationError
        invalid_user_mod = ModerationAction(
            admin_user=admin_user,
            action_type="user_block",
            target_user=None,
        )
        with self.assertRaises(ValidationError):
            invalid_user_mod.clean()

        # User action with target_user is valid
        valid_user_mod = ModerationAction(
            admin_user=admin_user,
            action_type="user_warning",
            target_user=target_user,
        )
        try:
            valid_user_mod.clean()
        except ValidationError as e:
            self.assertNotIn("target_user", str(e))

        # 2. Event action without event MUST raise ValidationError
        invalid_event_mod = ModerationAction(
            admin_user=admin_user,
            action_type="event_cancel",
            event=None,
        )
        with self.assertRaises(ValidationError):
            invalid_event_mod.clean()

        # Event action with event is valid
        valid_event_mod = ModerationAction(
            admin_user=admin_user,
            action_type="event_delete",
            event=event,
        )
        try:
            valid_event_mod.clean()
        except ValidationError as e:
            self.assertNotIn("event", str(e))

        # 3. Feedback action without feedback MUST raise ValidationError
        invalid_feedback_mod = ModerationAction(
            admin_user=admin_user,
            action_type="feedback_hide",
            feedback=None,
        )
        with self.assertRaises(ValidationError):
            invalid_feedback_mod.clean()

        # Feedback action with feedback is valid
        valid_feedback_mod = ModerationAction(
            admin_user=admin_user,
            action_type="feedback_hide",
            feedback=feedback,
        )
        try:
            valid_feedback_mod.clean()
        except ValidationError as e:
            self.assertNotIn("feedback", str(e))

        # 4. Verify CheckConstraint in Meta
        subject_constraint = next(
            c for c in ModerationAction._meta.constraints
            if c.name == "ck_moderation_subject"
        )
        self.assertIsNotNone(subject_constraint)

    def test_moderation_action_expiry_only_on_user_block(self):
        admin_user = self._create_dummy_user(99)
        target_user = self._create_dummy_user(10)
        event = Event(id=1, title="Test Event")
        now = timezone.now()

        # 1. user_block with expires_at is valid
        valid_expiring_block = ModerationAction(
            admin_user=admin_user,
            action_type="user_block",
            target_user=target_user,
            expires_at=now + timedelta(days=7),
        )
        try:
            valid_expiring_block.clean()
        except ValidationError as e:
            self.assertNotIn("expires_at", str(e))

        # 2. user_warning with expires_at MUST raise ValidationError
        invalid_warning = ModerationAction(
            admin_user=admin_user,
            action_type="user_warning",
            target_user=target_user,
            expires_at=now + timedelta(days=7),
        )
        with self.assertRaises(ValidationError):
            invalid_warning.clean()

        # 3. event_cancel with expires_at MUST raise ValidationError
        invalid_event_cancel = ModerationAction(
            admin_user=admin_user,
            action_type="event_cancel",
            event=event,
            expires_at=now + timedelta(days=7),
        )
        with self.assertRaises(ValidationError):
            invalid_event_cancel.clean()

        # 4. Verify CheckConstraint in Meta
        expiry_constraint = next(
            c for c in ModerationAction._meta.constraints
            if c.name == "ck_moderation_expiry"
        )
        self.assertEqual(
            expiry_constraint.condition,
            Q(expires_at__isnull=True) | Q(action_type="user_block"),
        )

# apps/audit/tests/test_moderation.py
from datetime import timedelta

from apps.common.exceptions import PermissionDeniedError

class ModerationServicesTests(TestCase):
    def setUp(self):
        # 1. Setup Admin
        self.admin = User.objects.create(
            phone_hash="hash_admin_mod",
            display_name="Super Admin",
            is_admin=True,
        )

        # 2. Setup regular user
        self.user = User.objects.create(
            phone_hash="hash_regular_user",
            display_name="Regular Volunteer",
            account_status="active",
        )

        # 3. Setup coordinator & event
        self.coordinator = User.objects.create(
            phone_hash="hash_coord_mod",
            display_name="Coordinator Mod",
            verification_status=VERIFICATION_VERIFIED,
        )
        CoordinatorProfile.objects.create(
            user=self.coordinator,
            status=COORDINATOR_APPROVED,
            approved_at=timezone.now(),
        )

        now = timezone.now()
        self.event = create_event(
            self.coordinator,
            {
                "title": "Clean River Event",
                "start_at": now + timedelta(days=3),
                "end_at": now + timedelta(days=3, hours=5),
            },
        )
        self.post = add_post(
            self.coordinator,
            self.event,
            {"title": "Water Collector", "required_headcount": 5},
        )
        publish_event(self.coordinator, self.event)

    def test_block_user_updates_state_and_writes_moderation_and_audit(self):
        expires_at = timezone.now() + timedelta(days=7)
        mod_action = block_user(
            admin=self.admin,
            target_user=self.user,
            reason="Violation of safety policies.",
            expires_at=expires_at,
        )

        # Check user state
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "blocked")
        self.assertIsNotNone(self.user.blocked_until)

        # Check moderation action
        self.assertEqual(mod_action.action_type, "user_block")
        self.assertEqual(mod_action.admin_user, self.admin)
        self.assertEqual(mod_action.target_user, self.user)
        self.assertEqual(mod_action.reason, "Violation of safety policies.")

        # Check audit log written
        audit = AuditLog.objects.filter(entity_type="user", entity_id=self.user.id, action="block_user").first()
        self.assertIsNotNone(audit)
        self.assertEqual(audit.actor_user, self.admin)
        self.assertEqual(audit.after_data["account_status"], "blocked")

    def test_delete_event_sets_soft_delete_and_hides_from_queries(self):
        mod_action = delete_event(
            admin=self.admin,
            event=self.event,
            reason="Event violates terms of service.",
        )

        self.event.refresh_from_db()
        self.assertIsNotNone(self.event.deleted_at)
        self.assertEqual(self.event.deleted_by, self.admin)

        # Event visible() queryset hides soft-deleted events
        visible_events = Event.objects.visible()
        self.assertNotIn(self.event, visible_events)

        # Restore event
        restore_event(admin=self.admin, event=self.event, reason="Resolved appeal.")
        self.event.refresh_from_db()
        self.assertIsNone(self.event.deleted_at)
        self.assertIsNone(self.event.deleted_by)
        self.assertIn(self.event, Event.objects.visible())

    def test_non_admin_cannot_execute_moderation(self):
        with self.assertRaises(PermissionDeniedError):
            block_user(admin=self.user, target_user=self.coordinator, reason="Unauthorized block")

        with self.assertRaises(PermissionDeniedError):
            delete_event(admin=self.user, event=self.event, reason="Unauthorized delete")

        with self.assertRaises(PermissionDeniedError):
            warn_user(admin=self.user, target_user=self.coordinator, reason="Unauthorized warn")

    def test_user_lifecycle_moderations(self):
        # 1. Warn
        warn = warn_user(self.admin, self.user, reason="Minor warning.")
        self.assertEqual(warn.action_type, "user_warning")

        # 2. Block and Unblock
        block_user(self.admin, self.user, reason="Temp block.")
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "blocked")

        unblock_user(self.admin, self.user, reason="Appeal granted.")
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "active")

        # 3. Close and Reinstate
        close_account(self.admin, self.user, reason="Permanent violation.")
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "closed")

        reinstate_account(self.admin, self.user, reason="Reinstated after review.")
        self.user.refresh_from_db()
        self.assertEqual(self.user.account_status, "active")

    def test_hide_feedback_and_dashboard_stats(self):
        app = EventApplication.objects.create(
            event=self.event,
            post=self.post,
            user=self.user,
            status="accepted",
            decided_at=timezone.now(),
        )
        create_assignment_from_application(app)
        activate_event(self.coordinator, self.event)
        complete_event(self.coordinator, self.event)

        fb = submit_feedback(
            user=self.user,
            event=self.event,
            direction=FEEDBACK_VOLUNTEER_TO_EVENT,
            score=1,
            comment="Inappropriate comment content.",
        )

        mod = hide_feedback(self.admin, fb, reason="Contains abusive language.")
        self.assertEqual(mod.action_type, "feedback_hide")
        fb.refresh_from_db()
        self.assertEqual(fb.status, "hidden")

        stats = dashboard_stats()
        self.assertIn("active_users", stats)
        self.assertIn("blocked_users", stats)
        self.assertIn("events_completed", stats)
        self.assertGreaterEqual(stats["events_completed"], 1)

    def test_cancel_event_by_admin(self):
        from apps.audit.services import cancel_event_by_admin
        mod = cancel_event_by_admin(self.admin, self.event, reason="Severe flood warning.")
        self.assertEqual(mod.action_type, "event_cancel")
        self.assertEqual(mod.admin_user, self.admin)
        self.event.refresh_from_db()
        self.assertEqual(self.event.status, "cancelled")
