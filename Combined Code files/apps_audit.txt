from django.core.exceptions import ValidationError
from django.db import models
from django.db.models import Q


class ModerationAction(models.Model):
    USER_ACTION_TYPES = [
        "user_warning",
        "user_block",
        "user_unblock",
        "account_close",
        "account_reinstate",
    ]
    EVENT_ACTION_TYPES = [
        "event_delete",
        "event_restore",
        "event_cancel",
    ]
    FEEDBACK_ACTION_TYPES = [
        "feedback_hide",
    ]

    ACTION_TYPE_CHOICES = [
        ("user_warning", "User Warning"),
        ("user_block", "User Block"),
        ("user_unblock", "User Unblock"),
        ("account_close", "Account Close"),
        ("account_reinstate", "Account Reinstate"),
        ("event_delete", "Event Delete"),
        ("event_restore", "Event Restore"),
        ("event_cancel", "Event Cancel"),
        ("feedback_hide", "Feedback Hide"),
    ]

    id = models.BigAutoField(primary_key=True)
    admin_user = models.ForeignKey(
        "accounts.User",
        on_delete=models.PROTECT,
        related_name="performed_moderations",
    )
    target_user = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="moderation_records",
    )
    event = models.ForeignKey(
        "events.Event",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="moderation_records",
    )
    feedback = models.ForeignKey(
        "ops.Feedback",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="moderation_records",
    )

    action_type = models.CharField(max_length=30, choices=ACTION_TYPE_CHOICES)
    reason = models.TextField(null=True, blank=True)
    expires_at = models.DateTimeField(null=True, blank=True)

    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = "moderation_actions"
        verbose_name = "Moderation Action"
        verbose_name_plural = "Moderation Actions"
        constraints = [
            models.CheckConstraint(
                condition=(
                    Q(
                        action_type__in=[
                            "user_warning",
                            "user_block",
                            "user_unblock",
                            "account_close",
                            "account_reinstate",
                        ]
                    )
                    & Q(target_user__isnull=False)
                )
                | (
                    Q(
                        action_type__in=[
                            "event_delete",
                            "event_restore",
                            "event_cancel",
                        ]
                    )
                    & Q(event__isnull=False)
                )
                | (
                    Q(action_type="feedback_hide")
                    & Q(feedback__isnull=False)
                ),
                name="ck_moderation_subject",
            ),
            models.CheckConstraint(
                condition=Q(expires_at__isnull=True) | Q(action_type="user_block"),
                name="ck_moderation_expiry",
            ),
        ]
        indexes = [
            models.Index(fields=["admin_user"], name="idx_mod_action_admin"),
            models.Index(fields=["target_user", "action_type"], name="idx_mod_action_target"),
            models.Index(fields=["event"], name="idx_mod_action_event"),
        ]

    def __str__(self):
        return f"{self.action_type} by Admin {self.admin_user_id}"

    def clean(self):
        super().clean()
        if self.action_type in self.USER_ACTION_TYPES and self.target_user is None:
            raise ValidationError({"target_user": f"target_user is required for '{self.action_type}'."})
        if self.action_type in self.EVENT_ACTION_TYPES and self.event is None:
            raise ValidationError({"event": f"event is required for '{self.action_type}'."})
        if self.action_type in self.FEEDBACK_ACTION_TYPES and self.feedback is None:
            raise ValidationError({"feedback": f"feedback is required for '{self.action_type}'."})
        if self.expires_at is not None and self.action_type != "user_block":
            raise ValidationError({"expires_at": "expires_at may only be set for 'user_block' action."})


class AuditLog(models.Model):
    id = models.BigAutoField(primary_key=True)
    actor_user = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="audit_logs",
    )
    action = models.CharField(max_length=100)
    entity_type = models.CharField(max_length=100)
    entity_id = models.BigIntegerField(null=True, blank=True)

    before_data = models.JSONField(null=True, blank=True)
    after_data = models.JSONField(null=True, blank=True)
    ip_address = models.GenericIPAddressField(null=True, blank=True)

    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = "audit_logs"
        verbose_name = "Audit Log"
        verbose_name_plural = "Audit Logs"
        indexes = [
            models.Index(fields=["entity_type", "entity_id"], name="idx_audit_entity"),
            models.Index(fields=["actor_user"], name="idx_audit_actor"),
            models.Index(fields=["created_at"], name="idx_audit_created"),
        ]

    def __str__(self):
        return f"{self.action} on {self.entity_type} #{self.entity_id}"

    def save(self, *args, **kwargs):
        if self.pk is not None and not getattr(self, "_state", None).adding:
            raise ValidationError("AuditLog records are append-only and cannot be updated.")
        super().save(*args, **kwargs)

    def delete(self, *args, **kwargs):
        raise ValidationError("AuditLog records are append-only and cannot be deleted.")

from datetime import timedelta
from django.db import transaction
from django.utils import timezone

from apps.accounts.models import CoordinatorProfile, User, UserVerificationImage
from apps.audit.models import AuditLog, ModerationAction
from apps.common.permissions import ensure_admin
from apps.events.models import Event
from apps.ops.models import Feedback


def log_action(
    actor=None,
    action: str = "",
    entity_type: str = "",
    entity_id=None,
    before=None,
    after=None,
    before_data=None,
    after_data=None,
    ip=None,
    request=None,
) -> AuditLog:
    ip_addr = ip
    if not ip_addr and request:
        ip_addr = request.META.get("REMOTE_ADDR")

    actor_user = actor if (actor and getattr(actor, "is_authenticated", False)) else None

    b_data = before if before is not None else before_data
    a_data = after if after is not None else after_data

    return AuditLog.objects.create(
        actor_user=actor_user,
        action=action,
        entity_type=entity_type,
        entity_id=entity_id,
        before_data=b_data,
        after_data=a_data,
        ip_address=ip_addr,
    )


def warn_user(admin, target_user: User, reason: str) -> ModerationAction:
    ensure_admin(admin)

    with transaction.atomic():
        action = ModerationAction.objects.create(
            admin_user=admin,
            target_user=target_user,
            action_type="user_warning",
            reason=reason,
        )

        log_action(
            actor=admin,
            action="warn_user",
            entity_type="user",
            entity_id=target_user.id,
            after_data={"reason": reason, "action_id": action.id},
        )

    return action


def block_user(admin, target_user: User, reason: str, expires_at=None) -> ModerationAction:
    ensure_admin(admin)

    with transaction.atomic():
        before = {
            "account_status": target_user.account_status,
            "blocked_until": str(target_user.blocked_until),
        }

        target_user.account_status = "blocked"
        target_user.blocked_until = expires_at
        target_user.save(update_fields=["account_status", "blocked_until"])

        action = ModerationAction.objects.create(
            admin_user=admin,
            target_user=target_user,
            action_type="user_block",
            reason=reason,
            expires_at=expires_at,
        )

        log_action(
            actor=admin,
            action="block_user",
            entity_type="user",
            entity_id=target_user.id,
            before_data=before,
            after_data={
                "account_status": "blocked",
                "blocked_until": str(expires_at),
                "action_id": action.id,
            },
        )

    return action


def unblock_user(admin, target_user: User, reason: str) -> ModerationAction:
    ensure_admin(admin)

    with transaction.atomic():
        before = {
            "account_status": target_user.account_status,

        }

        target_user.account_status = "active"

        action = ModerationAction.objects.create(
            admin_user=admin,
            target_user=target_user,
            action_type="user_unblock",
            reason=reason,
        )

        log_action(

            action="unblock_user",
            entity_type="user",
            entity_id=target_user.id,
            before_data=before,
            after_data={
                "account_status": "active",
                "blocked_until": None,
                "action_id": action.id,
            },
        )

    return action

def delete_event(admin, event: Event, reason: str) -> ModerationAction:
    ensure_admin(admin)

    with transaction.atomic():
        now = timezone.now()
        event.deleted_at = now
        event.deleted_by = admin
        event.save(update_fields=["deleted_at", "deleted_by"])

        action = ModerationAction.objects.create(
            admin_user=admin,
            event=event,
            action_type="event_delete",
            reason=reason,
        )

        log_action(
            actor=admin,
            action="delete_event",
            entity_type="event",
            entity_id=event.id,
            after_data={"deleted_at": str(now), "action_id": action.id},
        )

    return action


def restore_event(admin, event: Event, reason: str) -> ModerationAction:
    ensure_admin(admin)

    with transaction.atomic():
        event.deleted_at = None
        event.deleted_by = None
        event.save(update_fields=["deleted_at", "deleted_by"])

        action = ModerationAction.objects.create(
            admin_user=admin,
            event=event,
            action_type="event_restore",
            reason=reason,
        )

        log_action(
            actor=admin,
            action="restore_event",
            entity_type="event",
            entity_id=event.id,
            after_data={"restored": True, "action_id": action.id},
        )

    return action


def cancel_event_by_admin(admin, event: Event, reason: str) -> ModerationAction:
    ensure_admin(admin)

    from apps.events.services import cancel_event

    with transaction.atomic():
        cancel_event(actor=admin, event=event, reason=reason)

        action = ModerationAction.objects.create(
            admin_user=admin,
            event=event,
            action_type="event_cancel",
            reason=reason,
        )

        log_action(
            actor=admin,
            action="cancel_event_by_admin",
            entity_type="event",
            entity_id=event.id,
            after_data={"reason": reason, "action_id": action.id},
        )

    return action

def dashboard_stats() -> dict:
    thirty_days_ago = timezone.now() - timedelta(days=30)
    return {
        "active_users": User.objects.filter(account_status="active").count(),
        "blocked_users": User.objects.filter(account_status="blocked").count(),
        "users_pending_verification": User.objects.filter(verification_status="pending").count(),
        "images_pending_review": UserVerificationImage.objects.filter(status="pending", is_current=True).count(),
        "coordinators_pending_approval": CoordinatorProfile.objects.filter(status="pending").count(),
        "coordinators_approved": CoordinatorProfile.objects.filter(status="approved").count(),
        "events_draft": Event.objects.filter(deleted_at__isnull=True, status="draft").count(),
        "events_published": Event.objects.filter(deleted_at__isnull=True, status="published").count(),
        "events_active": Event.objects.filter(deleted_at__isnull=True, status="active").count(),
        "events_completed": Event.objects.filter(deleted_at__isnull=True, status="completed").count(),
        "events_cancelled": Event.objects.filter(deleted_at__isnull=True, status="cancelled").count(),
        "events_deleted": Event.objects.filter(deleted_at__isnull=False).count(),
        "warnings_last_30d": ModerationAction.objects.filter(
            action_type="user_warning",
            created_at__gte=thirty_days_ago,
        ).count(),
    }

class AdminWarnUserView(APIView):
    permission_classes = [IsAuthenticated, IsAdminRole]

    def post(self, request, pk):
        target_user = get_object_or_404(User, pk=pk)
        serializer = ModerationReasonSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        action = warn_user(
            admin=request.user,
