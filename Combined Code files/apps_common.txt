# apps/common/exceptions.py
from rest_framework import status
from rest_framework.response import Response


class DomainError(Exception):
    code = "domain_error"
    message = "A domain error occurred."
    http_status = status.HTTP_400_BAD_REQUEST

    def __init__(self, message=None, code=None, http_status=None):
        self.message = message or self.message
        self.code = code or self.code
        self.http_status = http_status or self.http_status
        super().__init__(self.message)


class PermissionDeniedError(DomainError):
    code = "permission_denied"
    message = "You do not have permission to perform this action."
    http_status = status.HTTP_403_FORBIDDEN


class AccountBlockedError(DomainError):
    code = "account_blocked"
    message = "This account is blocked."
    http_status = status.HTTP_403_FORBIDDEN


class AccountClosedError(DomainError):
    code = "account_closed"
    message = "This account is closed."
    http_status = status.HTTP_403_FORBIDDEN


class InvalidStateError(DomainError):
    code = "invalid_state"
    message = "This action is not allowed in the current state."
    http_status = status.HTTP_409_CONFLICT


class CapacityFullError(DomainError):
    code = "capacity_full"
    message = "No remaining capacity."
    http_status = status.HTTP_409_CONFLICT


class OnboardingRequiredError(DomainError):
    code = "onboarding_required"
    message = "Onboarding must be acknowledged before this action."
    http_status = status.HTTP_409_CONFLICT


class ApplicationNotAllowedError(DomainError):
    code = "application_not_allowed"
    message = "Application is not allowed."
    http_status = status.HTTP_400_BAD_REQUEST


class QrTokenInvalidError(DomainError):
    code = "qr_token_invalid"
    message = "QR token is invalid."
    http_status = status.HTTP_400_BAD_REQUEST


class NotFoundError(DomainError):
    code = "not_found"
    message = "Resource not found."
    http_status = status.HTTP_404_NOT_FOUND


class InvalidOtpError(DomainError):
    code = "invalid_otp"
    message = "Invalid OTP."
    http_status = status.HTTP_400_BAD_REQUEST


class TooManyAttemptsError(DomainError):
    code = "too_many_attempts"
    message = "Too many attempts."
    http_status = status.HTTP_429_TOO_MANY_REQUESTS


class VerificationRequiredError(DomainError):
    code = "verification_required"
    message = "Identity verification is required."
    http_status = status.HTTP_409_CONFLICT


def domain_exception_handler(exc, context):
    if isinstance(exc, DomainError):
        return Response(
            {
                "error": {
                    "code": exc.code,
                    "message": exc.message,
                }
            },
            status=exc.http_status,
        )

    from rest_framework.views import exception_handler
    return exception_handler(exc, context)

# apps/common/fields.py
from django.db import models


class ChoiceArrayField(models.JSONField):
    """
    Portable Array/List field using JSONField.
    Stores arrays as JSON lists, works uniformly across SQLite and PostgreSQL.
    """
    def __init__(self, base_field=None, **kwargs):
        kwargs.pop("size", None)
        kwargs.setdefault("default", list)
        self.base_field = base_field
        super().__init__(**kwargs)

    def deconstruct(self):
        name, path, args, kwargs = super().deconstruct()
        if self.base_field:
            kwargs["base_field"] = self.base_field
        return name, path, args, kwargs

# apps/common/permissions.py
from django.utils import timezone
from rest_framework.permissions import BasePermission

from apps.common.constants import (
    ACCOUNT_BLOCKED,
    ACCOUNT_CLOSED,
    COORDINATOR_APPROVED,
)
from apps.common.exceptions import (
    AccountBlockedError,
    AccountClosedError,
    PermissionDeniedError,
)


def is_admin(user) -> bool:
    return bool(user and user.is_authenticated and (getattr(user, "is_admin", False) or getattr(user, "is_superuser", False)))


def is_approved_coordinator(user) -> bool:
    profile = getattr(user, "coordinator_profile", None)
    return bool(profile and getattr(profile, "status", None) == COORDINATOR_APPROVED)


def is_event_manager(user, event) -> bool:
    if not (user and user.is_authenticated):
        return False
    if is_admin(user):
        return True
    coordinator_id = getattr(event, "coordinator_id", None)
    if coordinator_id is None and hasattr(event, "coordinator"):
        coordinator_id = getattr(event.coordinator, "user_id", getattr(event.coordinator, "id", None))
    return bool(coordinator_id == getattr(user, "id", None))


def ensure_active_account(user):
    if not user or not user.is_authenticated:
        raise PermissionDeniedError("Authentication is required.")

    if getattr(user, "account_status", None) == ACCOUNT_CLOSED:
        raise AccountClosedError()

    if getattr(user, "account_status", None) == ACCOUNT_BLOCKED:
        blocked_until = getattr(user, "blocked_until", None)
        if blocked_until is None or blocked_until > timezone.now():
            raise AccountBlockedError()
        # Expired block is treated as unblocked for request purposes.

    return user


def ensure_admin(user):
    ensure_active_account(user)
    if not is_admin(user):
        raise PermissionDeniedError("Admin permission is required.")
    return user


def ensure_approved_coordinator(user):
    ensure_active_account(user)
    if not is_approved_coordinator(user):
        raise PermissionDeniedError("Approved coordinator permission is required.")
    return user


def ensure_event_manager(user, event):
    ensure_active_account(user)
    if not is_event_manager(user, event):
        raise PermissionDeniedError("You are not a manager of this event.")
    return user


class IsActiveAccount(BasePermission):
    def has_permission(self, request, view):
        try:
            ensure_active_account(request.user)
            return True
        except Exception:
            return False


class IsAdminRole(BasePermission):
    def has_permission(self, request, view):
        try:
            ensure_admin(request.user)
            return True
        except Exception:
            return False


class IsApprovedCoordinatorRole(BasePermission):
    def has_permission(self, request, view):
        try:
            ensure_approved_coordinator(request.user)
            return True
        except Exception:
            return False

# apps/common/security.py
import base64
import hashlib
import hmac
import re
import secrets

from cryptography.fernet import Fernet
from django.conf import settings


def _secret() -> bytes:
    secret = getattr(settings, "PHONE_HASH_HMAC_SECRET", None)
    if not secret:
        raise RuntimeError("PHONE_HASH_HMAC_SECRET is not configured.")
    return str(secret).encode("utf-8")


def normalize_phone(raw_phone: str) -> str:
    """
    Minimal phone normalization.
    Keep leading + if present, otherwise digits only.
    """
    raw_phone = (raw_phone or "").strip()
    if raw_phone.startswith("+"):
        digits = re.sub(r"[^\d]", "", raw_phone[1:])
        return f"+{digits}"
    digits = re.sub(r"[^\d]", "", raw_phone)
    return digits


def compute_phone_hash(raw_phone: str) -> str:
    normalized = normalize_phone(raw_phone)
    return hmac.new(
        _secret(),
        normalized.encode("utf-8"),
        hashlib.sha256,
    ).hexdigest()


def _fernet() -> Fernet:
    key_material = hashlib.sha256(_secret()).digest()
    key = base64.urlsafe_b64encode(key_material)
    return Fernet(key)


def encrypt_phone(raw_phone: str) -> str:
    normalized = normalize_phone(raw_phone)
    return _fernet().encrypt(normalized.encode("utf-8")).decode("utf-8")


def decrypt_phone(encrypted_phone: str) -> str:
    return _fernet().decrypt(encrypted_phone.encode("utf-8")).decode("utf-8")


def generate_otp_code() -> str:
    return f"{secrets.randbelow(1_000_000):06d}"


def hash_otp(raw_code: str) -> str:
    return hmac.new(
        _secret(),
        str(raw_code).encode("utf-8"),
        hashlib.sha256,
    ).hexdigest()


def generate_raw_token() -> str:
    return secrets.token_urlsafe(32)


def hash_token(raw_token: str) -> str:
    return hashlib.sha256(raw_token.encode("utf-8")).hexdigest()

# apps/common/validators.py
import hashlib
import os
import uuid
from io import BytesIO
from pathlib import Path
from PIL import Image
from django.conf import settings

from apps.common.exceptions import DomainError


def validate_and_transcode_jpeg(uploaded_file, destination_dir: str | Path) -> tuple[str, str]:
    """
    Open image using Pillow, verify, strip EXIF, convert to RGB,
    save to destination_dir as JPEG, return (relative_file_path, sha256_hex).
    """
    try:
        img = Image.open(uploaded_file)
        # Ensure image is loaded
        img.load()
        if img.mode in ("RGBA", "LA"):
            background = Image.new("RGB", img.size, (255, 255, 255))
            background.paste(img, mask=img.split()[-1])
            rgb_image = background
        else:
            rgb_image = img.convert("RGB")
    except Exception as e:
        raise DomainError(f"Invalid or corrupted image: {str(e)}", code="invalid_image")

    dest_path = Path(destination_dir)
    dest_path.mkdir(parents=True, exist_ok=True)

    filename = f"{uuid.uuid4().hex}.jpg"
    full_path = dest_path / filename

    # Save to buffer to strip EXIF and get SHA-256
    buffer = BytesIO()
    rgb_image.save(buffer, format="JPEG", quality=90, optimize=True)
    image_bytes = buffer.getvalue()

    sha256_hex = hashlib.sha256(image_bytes).hexdigest()

    with open(full_path, "wb") as f:
        f.write(image_bytes)

    # Determine relative path
    media_root = getattr(settings, "MEDIA_ROOT", None)
    if media_root:
        try:
            rel_path = full_path.relative_to(Path(media_root)).as_posix()
        except ValueError:
            rel_path = f"{dest_path.name}/{filename}"
    else:
        rel_path = f"{dest_path.name}/{filename}"

    return rel_path, sha256_hex
