class AccountsApiTests(APITestCase):

        # 1. Request OTP
        req_resp = self.client.post("/api/v1/auth/otp/request/", {"phone": self.phone})
        self.assertEqual(req_resp.status_code, status.HTTP_200_OK)
        self.assertEqual(req_resp.data.get("status"), "sent")
        code = req_resp.data.get("dev_code")
        self.assertIsNotNone(code)

        # 2. Verify OTP
        verify_resp = self.client.post("/api/v1/auth/otp/verify/", {"phone": self.phone, "code": code})

        self.assertIn("token", verify_resp.data)
        self.assertIn("user", verify_resp.data)
        self.assertTrue(verify_resp.data.get("created"))

        user = User.objects.create(
            phone_hash="hash_user_patch",
            display_name="Regular Volunteer",
            is_admin=False,
        )
        token = issue_token_for_user(user)
        self.client.credentials(HTTP_AUTHORIZATION=f"Token {token.key}")

        # Attempt to patch regular fields - should succeed
        patch_resp = self.client.patch("/api/v1/me/", {"display_name": "Updated Name", "home_area": "Yangon"}, format="json")

        self.assertEqual(elevate_resp.data["error"]["code"], "field_not_allowed")

from datetime import timedelta
from django.test import TestCase
from django.utils import timezone
from rest_framework.authtoken.models import Token

from apps.accounts.authentication import BlockedAwareTokenAuthentication
from apps.accounts.models import AuthOtp, User

from apps.common.constants import ACCOUNT_ACTIVE, ACCOUNT_BLOCKED, ACCOUNT_CLOSED
from apps.common.exceptions import (
    AccountBlockedError,
    AccountClosedError,
    InvalidOtpError,
    TooManyAttemptsError,
)
from apps.common.security import compute_phone_hash, hash_otp


class AuthServicesTests(TestCase):
    def setUp(self):
        self.phone = "+959123456789"
        self.phone_hash = compute_phone_hash(self.phone)

    def test_otp_request_and_verify_creates_user(self):
        # Request OTP
        req_res = request_otp(self.phone, purpose="login")
        self.assertEqual(req_res.get("status"), "sent")
        code = req_res.get("dev_code")
        self.assertIsNotNone(code)

        # Verify OTP creates user and token
        user, token, created = verify_otp(self.phone, code, purpose="login")
        self.assertTrue(created)
        self.assertEqual(user.phone_hash, self.phone_hash)
        self.assertEqual(user.account_status, ACCOUNT_ACTIVE)
        self.assertIsNotNone(token)
        self.assertEqual(token.user, user)

        # Second verify for same user should not create a new user
        req_res2 = request_otp(self.phone, purpose="login")
        code2 = req_res2.get("dev_code")
        user2, token2, created2 = verify_otp(self.phone, code2, purpose="login")
        self.assertFalse(created2)
        self.assertEqual(user2.id, user.id)
        self.assertEqual(token2.key, token.key)

    def test_otp_reuse_fails(self):
        req_res = request_otp(self.phone, purpose="login")
        code = req_res.get("dev_code")

        # First verification succeeds
        verify_otp(self.phone, code, purpose="login")

        # Reusing the consumed OTP must raise InvalidOtpError
        with self.assertRaises(InvalidOtpError):
            verify_otp(self.phone, code, purpose="login")

    def test_otp_expired_fails(self):
        code = "123456"
        # Create an expired OTP directly
        AuthOtp.objects.create(
            phone_hash=self.phone_hash,
            code_hash=hash_otp(code),
            purpose="login",
            expires_at=timezone.now() - timedelta(seconds=10),
        )

        with self.assertRaises(InvalidOtpError):
            verify_otp(self.phone, code, purpose="login")

    def test_otp_max_attempts_locks(self):
        req_res = request_otp(self.phone, purpose="login")
        code = req_res.get("dev_code")
        wrong_code = "000000" if code != "000000" else "111111"

        # Fail 5 times with wrong code
        for _ in range(5):
            with self.assertRaises(InvalidOtpError):
                verify_otp(self.phone, wrong_code, purpose="login")

        # 6th attempt with correct code should fail with TooManyAttemptsError
        with self.assertRaises(TooManyAttemptsError):
            verify_otp(self.phone, code, purpose="login")

    def test_blocked_user_rejected_by_auth_class(self):
        # Create a blocked user and token
        user = User.objects.create(
            phone_hash=self.phone_hash,
            display_name="Blocked User",
            account_status=ACCOUNT_BLOCKED,
            blocked_until=timezone.now() + timedelta(days=1),
        )
        token = Token.objects.create(user=user)

        authenticator = BlockedAwareTokenAuthentication()

        with self.assertRaises(AccountBlockedError):
            authenticator.authenticate_credentials(token.key)

    def test_closed_account_cannot_request_otp(self):
        User.objects.create(
            phone_hash=self.phone_hash,
            display_name="Closed User",
            account_status=ACCOUNT_CLOSED,
        )
        with self.assertRaises(AccountClosedError):
            request_otp(self.phone, purpose="login")

class AccountsModelConstraintTests(SimpleTestCase):
    def test_blocked_until_constraint_violation(self):
        # 1. Active account with blocked_until MUST raise ValidationError
        invalid_user = User(
            phone_hash="hash_blocked_until_violation",
            phone_enc="enc_phone",
            display_name="Invalid Blocked User",
            account_status=ACCOUNT_ACTIVE,
            blocked_until=timezone.now() + timedelta(days=1),
        )
        with self.assertRaises(ValidationError):
            invalid_user.clean()

        # 2. Blocked account with blocked_until is valid
        valid_blocked_user = User(
            phone_hash="hash_valid_blocked",
            phone_enc="enc_phone",
            display_name="Valid Blocked User",
            account_status=ACCOUNT_BLOCKED,
            blocked_until=timezone.now() + timedelta(days=1),
        )
        # Should not raise validation error for blocked_until
        try:
            valid_blocked_user.clean()
        except ValidationError as e:
            self.assertNotIn("blocked_until", e.message_dict)

        # 3. Active account with blocked_until=None is valid
        valid_active_user = User(
            phone_hash="hash_valid_active",
            phone_enc="enc_phone",
            display_name="Valid Active User",
            account_status=ACCOUNT_ACTIVE,
            blocked_until=None,
        )
        try:
            valid_active_user.clean()
        except ValidationError as e:
            self.assertNotIn("blocked_until", e.message_dict)

        # Verify the CheckConstraint condition in Meta
        constraint = next(
            c for c in User._meta.constraints if c.name == "ck_users_blocked_until"
        )
        self.assertIsNotNone(constraint)
        self.assertEqual(
            constraint.condition,
            Q(blocked_until__isnull=True) | Q(account_status="blocked"),
        )

    def test_donation_consistency_constraint_violation(self):
        # 1. Willing to donate blood without is_willing_to_donate=True MUST fail
        invalid_blood_donor = User(
            phone_hash="hash_invalid_blood_donor",
            phone_enc="enc_phone",
            display_name="Invalid Blood Donor",
            is_willing_to_donate=False,
            is_willing_to_donate_blood=True,
            is_willing_to_donate_items=False,
        )
        with self.assertRaises(ValidationError):
            invalid_blood_donor.clean()

        # 2. Willing to donate items without is_willing_to_donate=True MUST fail
        invalid_items_donor = User(
            phone_hash="hash_invalid_items_donor",
            phone_enc="enc_phone",
            display_name="Invalid Items Donor",
            is_willing_to_donate=False,
            is_willing_to_donate_blood=False,
            is_willing_to_donate_items=True,
        )
        with self.assertRaises(ValidationError):
            invalid_items_donor.clean()

        # 3. Willing to donate=True with blood/items is valid
        valid_donor = User(
            phone_hash="hash_valid_donor",
            phone_enc="enc_phone",
            display_name="Valid Donor",
            is_willing_to_donate=True,
            is_willing_to_donate_blood=True,
            is_willing_to_donate_items=True,
        )
        try:
            valid_donor.clean()
        except ValidationError as e:
            self.assertNotIn("is_willing_to_donate", e.message_dict)

        # 4. Willing to donate=False with blood=False, items=False is valid
        valid_non_donor = User(
            phone_hash="hash_valid_non_donor",
            phone_enc="enc_phone",
            display_name="Valid Non Donor",
            is_willing_to_donate=False,
            is_willing_to_donate_blood=False,
            is_willing_to_donate_items=False,
        )
        try:
            valid_non_donor.clean()
        except ValidationError as e:
            self.assertNotIn("is_willing_to_donate", e.message_dict)

        # Verify the CheckConstraint condition in Meta
        constraint = next(
            c for c in User._meta.constraints if c.name == "ck_users_donation_consistency"
        )
        self.assertIsNotNone(constraint)
        self.assertEqual(
            constraint.condition,
            Q(is_willing_to_donate=True)
            | (Q(is_willing_to_donate_blood=False) & Q(is_willing_to_donate_items=False)),
        )

    def test_single_current_verification_image_constraint(self):
        # Verify UniqueConstraint specification on UserVerificationImage
        constraint = next(
            c for c in UserVerificationImage._meta.constraints
            if c.name == "uq_current_verification_image"
        )
        self.assertIsNotNone(constraint)
        self.assertEqual(list(constraint.fields), ["user", "image_type"])

            c for c in UserVerificationImage._meta.constraints

            review_constraint.condition,

    def test_verification_image_transcoded_to_jpeg_and_old_deactivated(self):

        file1 = create_dummy_uploaded_image("selfie1.png", format="PNG", color="blue")
        img1 = add_verification_image(self.user, IMAGE_MOCK_SELFIE, file1)

        self.assertTrue(img1.is_current)
        self.assertEqual(img1.file_format, "jpeg")
        self.assertTrue(img1.image_path.name.endswith(".jpg"))
        self.assertIsNotNone(img1.file_sha256)

        # User status should now be pending
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        file2 = create_dummy_uploaded_image("selfie2.png", format="PNG", color="green")

        # img1 must now be deactivated (is_current=False), img2 is current
        img1.refresh_from_db()
        self.assertFalse(img1.is_current)
        self.assertTrue(img2.is_current)
        self.assertNotEqual(img1.id, img2.id)

    def test_recompute_verification_status_logic(self):
        # Initially none
        self.assertEqual(self.user.verification_status, VERIFICATION_NONE)

        # Upload selfie -> pending
        file_selfie = create_dummy_uploaded_image("selfie.png")
        img_selfie = add_verification_image(self.user, IMAGE_MOCK_SELFIE, file_selfie)
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        # Upload ID -> pending
        file_id = create_dummy_uploaded_image("id.png")
        img_id = add_verification_image(self.user, IMAGE_ID_LICENSE, file_id)
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        # Approve selfie only -> still pending because ID is pending
        review_verification_image(self.admin, img_selfie, status="approved")
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        # Approve ID -> both approved -> verified!
        review_verification_image(self.admin, img_id, status="approved")
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_VERIFIED)

        # If a new ID is uploaded and rejected -> status becomes rejected
        file_id2 = create_dummy_uploaded_image("id2.png")
        img_id2 = add_verification_image(self.user, IMAGE_ID_LICENSE, file_id2)
        review_verification_image(self.admin, img_id2, status="rejected", note="Blurry image")
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_REJECTED)

    def test_coordinator_approval_requires_verified_user(self):
        # Apply as coordinator
        profile = apply_as_coordinator(self.user, notes="Ready to lead events")
        self.assertEqual(profile.status, COORDINATOR_PENDING)

        # Non-admin cannot approve
        non_admin = User.objects.create(phone_hash="hash_regular", display_name="Regular")
        with self.assertRaises(PermissionDeniedError):
            approve_coordinator(non_admin, profile)

        # Admin cannot approve if user is unverified

        with self.assertRaises(VerificationRequiredError):
            approve_coordinator(self.admin, profile)

        # Now verify user
        self.user.verification_status = VERIFICATION_VERIFIED
        self.user.save()

        # Approval succeeds
        approved_profile = approve_coordinator(self.admin, profile)
        self.assertEqual(approved_profile.status, COORDINATOR_APPROVED)

        self.assertEqual(suspended.status, COORDINATOR_SUSPENDED)

        # Revoke
        revoked = revoke_coordinator(self.admin, suspended)
