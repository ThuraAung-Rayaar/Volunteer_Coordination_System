@admin.register(Event)
class EventAdmin(admin.ModelAdmin):
    list_display = (
        "id",
        "title",
        "coordinator",
        "purpose_type",
        "status",
        "start_at",
        "end_at",
        "deleted_at",
    )
    list_filter = (
        "status",
        "purpose_type",
        "compensation_kind",
        "meals_provided",
        "lodging_provided",
    )
    search_fields = ("title", "description", "location_name")
    inlines = [EventCharityFieldInline, EventPostInline]
    actions = [
        "publish_event",
        "activate_event",
        "soft_delete_event",
        "restore_event",
        "publish_selected_events",
        "activate_selected_events",
        "soft_delete_selected_events",
        "restore_selected_events",
    ]

    @admin.action(description="Publish selected draft events")
    def publish_event(self, request, queryset):
        for event in queryset:
            if event.status == "draft":
                try:
                    publish_event(actor=request.user, event=event)
                except Exception as e:
                    self.message_user(request, f"Error publishing event #{event.id}: {e}", level="error")

    @admin.action(description="Activate selected published events")
    def activate_event(self, request, queryset):
        for event in queryset:
            if event.status == "published":
                try:
                    activate_event(actor=request.user, event=event)
                except Exception as e:
                    self.message_user(request, f"Error activating event #{event.id}: {e}", level="error")

    @admin.action(description="Soft delete selected events")
    def soft_delete_event(self, request, queryset):
        for event in queryset:
            if event.deleted_at is None:
                try:
                    delete_event(admin=request.user, event=event, reason="Deleted via admin panel")
                except Exception as e:
                    self.message_user(request, f"Error deleting event #{event.id}: {e}", level="error")

    @admin.action(description="Restore selected deleted events")
    def restore_event(self, request, queryset):
        for event in queryset:
            if event.deleted_at is not None:
                try:
                    restore_event(admin=request.user, event=event, reason="Restored via admin panel")
                except Exception as e:
                    self.message_user(request, f"Error restoring event #{event.id}: {e}", level="error")

    publish_selected_events = publish_event
    activate_selected_events = activate_event
    soft_delete_selected_events = soft_delete_event
    restore_selected_events = restore_event

class Event(models.Model):

        ("charity_event", "Charity Event"),
        ("help_wanted_event", "Help Wanted Event"),
    ]

    STATUS_CHOICES = [
        (EVENT_DRAFT, "Draft"),
        (EVENT_PUBLISHED, "Published"),
        (EVENT_ACTIVE, "Active"),
        (EVENT_COMPLETED, "Completed"),
        (EVENT_CANCELLED, "Cancelled"),

        ("first_experienced", "First Experienced"),

        ("unpaid", "Unpaid"),
        ("arranged_outside_system", "Arranged Outside System"),
    ]

    id = models.BigAutoField(primary_key=True)

    coordinator = models.ForeignKey(
        "accounts.User",
        on_delete=models.PROTECT,
        related_name="coordinated_events",

    title = models.CharField(max_length=255)
    description = models.TextField(null=True, blank=True)
    purpose_type = models.CharField(

        choices=PURPOSE_TYPE_CHOICES,
        default="charity_event",
    )

    location_name = models.CharField(max_length=255, null=True, blank=True)
    landmark_text = models.TextField(null=True, blank=True)

    start_at = models.DateTimeField(db_index=True)
    end_at = models.DateTimeField()
    application_deadline = models.DateTimeField(null=True, blank=True)

    status = models.CharField(
        max_length=20,
        choices=STATUS_CHOICES,
        default=EVENT_DRAFT,
    )

    acceptance_mode = models.CharField(
        max_length=20,
        choices=ACCEPTANCE_MODE_CHOICES,
        default="manual",
    )
    auto_assign_enabled = models.BooleanField(default=True)
    group_size = models.IntegerField(null=True, blank=True)
    leader_strategy = models.CharField(
        max_length=30,
        choices=LEADER_STRATEGY_CHOICES,
        default="none",
    )
    max_participants = models.IntegerField(null=True, blank=True)

    # Recruiting conditions (informational only)
    meals_provided = models.BooleanField(default=False)
    lodging_provided = models.BooleanField(default=False)
    travel_support_provided = models.BooleanField(default=False)

    compensation_kind = models.CharField(
        max_length=30,
        choices=COMPENSATION_KIND_CHOICES,
        default="unpaid",
    )
    compensation_note = models.TextField(null=True, blank=True)

    # Onboarding
    onboarding_text = models.TextField(null=True, blank=True)
    onboarding_version = models.IntegerField(default=1)
    require_onboarding_ack = models.BooleanField(default=True)

    # Walk-in QR registration
    qr_registration_enabled = models.BooleanField(default=True)

    # Soft delete
    deleted_at = models.DateTimeField(null=True, blank=True)
    deleted_by = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="deleted_events",
    )

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    objects = EventQuerySet.as_manager()

    class Meta:
        db_table = "events"
        verbose_name = "Event"
        verbose_name_plural = "Events"
        constraints = [
            models.CheckConstraint(
                condition=Q(end_at__gt=F("start_at")),
                name="ck_events_time",
            ),
            models.CheckConstraint(
                condition=Q(application_deadline__isnull=True)
                | Q(application_deadline__lte=F("start_at")),
                name="ck_events_deadline",
            ),
            models.CheckConstraint(
                condition=(
                    Q(deleted_at__isnull=True) & Q(deleted_by__isnull=True)
                )
                | (
                    Q(deleted_at__isnull=False) & Q(deleted_by__isnull=False)
                ),
                name="ck_events_soft_delete",
            ),
            models.CheckConstraint(
                condition=Q(group_size__isnull=True) | Q(group_size__gt=0),
                name="ck_events_group_size",
            ),
            models.CheckConstraint(
                condition=Q(max_participants__isnull=True) | Q(max_participants__gt=0),
                name="ck_events_max_participants",
            ),
            models.CheckConstraint(
                condition=Q(onboarding_version__gte=1),
                name="ck_events_onboarding_version",
            ),
        ]
        indexes = [
            models.Index(fields=["coordinator"], name="idx_events_coordinator"),
            models.Index(fields=["start_at"], name="idx_events_start_at"),
            models.Index(
                fields=["status", "start_at"],
                name="idx_events_visible",
                condition=Q(deleted_at__isnull=True),
            ),
        ]

    def __str__(self):
        return f"{self.title} ({self.status})"

    def clean(self):
        super().clean()
        if self.start_at and self.end_at and self.end_at <= self.start_at:
            raise ValidationError({"end_at": "end_at must be greater than start_at."})
        if self.application_deadline and self.start_at and self.application_deadline > self.start_at:
            raise ValidationError({"application_deadline": "application_deadline must be <= start_at."})
        if (self.deleted_at is None) != (self.deleted_by is None):
            raise ValidationError("deleted_at and deleted_by must be set together or both null.")
        if self.group_size is not None and self.group_size <= 0:
            raise ValidationError({"group_size": "group_size must be greater than 0."})
        if self.max_participants is not None and self.max_participants <= 0:
            raise ValidationError({"max_participants": "max_participants must be greater than 0."})
        if self.onboarding_version is not None and self.onboarding_version < 1:
            raise ValidationError({"onboarding_version": "onboarding_version must be >= 1."})

def create_event(actor, data: dict, charity_codes: list[str] = ()) -> Event:
    """
    Coordinator or Admin creates a new event in draft status.
    """
    if not is_admin(actor):
        ensure_approved_coordinator(actor)

    clean_data = dict(data)
    # Ensure status starts at draft and coordinator is actor
    clean_data["status"] = EVENT_DRAFT
    clean_data["coordinator"] = actor
    clean_data.setdefault("onboarding_version", 1)

    # Pop charity codes if included in data dict
    codes = charity_codes or clean_data.pop("charity_codes", ())

    with transaction.atomic():
        event = Event.objects.create(**clean_data)
        if codes:
            set_event_charity_fields(actor, event, codes)

        log_action(
            actor=actor,
            action="create_event",
            entity_type="event",
            entity_id=event.id,
            after_data={"title": event.title, "start_at": str(event.start_at)},
        )

    return event


def update_event(actor, event: Event, data: dict) -> Event:
    """
    Coordinator or Admin updates event fields.
    If onboarding_text changes, onboarding_version is incremented automatically.
    """
    ensure_event_manager(actor, event)

    clean_data = dict(data)
    # Do not allow arbitrary status jumps via update_event
    clean_data.pop("status", None)
    clean_data.pop("coordinator", None)

    with transaction.atomic():
        # Check if onboarding_text changed
        if "onboarding_text" in clean_data:
            new_text = clean_data["onboarding_text"]
            if new_text != event.onboarding_text:
                event.onboarding_version += 1

        codes = clean_data.pop("charity_codes", None)
        if codes is not None:
            set_event_charity_fields(actor, event, codes)

        for field, value in clean_data.items():
            if hasattr(event, field):
                setattr(event, field, value)

        event.clean()
        event.save()

        log_action(
            actor=actor,
            action="update_event",
            entity_type="event",
            entity_id=event.id,
            after_data={k: str(v) for k, v in clean_data.items()},
        )

    return event


def publish_event(actor, event: Event) -> Event:
    """
    Transition event from draft to published.
    Requires at least one post.
    """
    ensure_event_manager(actor, event)

    if event.status != EVENT_DRAFT:
        raise InvalidStateError(f"Cannot publish event in status '{event.status}'. Must be 'draft'.")

    if not event.posts.exists():
        raise DomainError("Event must have at least one post before publishing.", code="post_required")

    with transaction.atomic():
        event.status = EVENT_PUBLISHED
        event.save(update_fields=["status"])

        log_action(
            actor=actor,
            action="publish_event",
            entity_type="event",
            entity_id=event.id,
        )

    return event

def complete_event(actor, event: Event) -> Event:
    """
    Transition event from active to completed.
    Cascades status:
    - Active/confirmed assignments become completed
    - Checked-in attendances become completed
    """
    ensure_event_manager(actor, event)

    if event.status != EVENT_ACTIVE:
        raise InvalidStateError(f"Cannot complete event in status '{event.status}'. Must be 'active'.")

    from apps.ops.models import Assignment, Attendance

    with transaction.atomic():
        event.status = EVENT_COMPLETED
        event.save(update_fields=["status"])

        # Complete active assignments
        Assignment.objects.filter(
            event=event,
            status__in=ACTIVE_ASSIGNMENT_STATUSES,
        ).update(status=ASSIGNMENT_COMPLETED)

        # Complete checked-in attendances
        Attendance.objects.filter(
            event=event,
            status=ATTENDANCE_CHECKED_IN,
        ).update(status=ATTENDANCE_COMPLETED)

        log_action(
            actor=actor,
            action="complete_event",
            entity_type="event",
            entity_id=event.id,
        )

    return event


def cancel_event(actor, event: Event, reason: str = "") -> Event:
    """
    Cancel an event (from published or active).
    Cascades:
    - Active assignments become cancelled
    - Expected attendances (not checked in) become absent
    """
    ensure_event_manager(actor, event)

    if event.status not in (EVENT_PUBLISHED, EVENT_ACTIVE):
        raise InvalidStateError(
            f"Cannot cancel event in status '{event.status}'. Must be 'published' or 'active'."
        )

    from apps.ops.models import Assignment, Attendance

    with transaction.atomic():
        event.status = EVENT_CANCELLED
        event.save(update_fields=["status"])

        # Cancel active assignments
        Assignment.objects.filter(
            event=event,
            status__in=ACTIVE_ASSIGNMENT_STATUSES,
        ).update(status=ASSIGNMENT_CANCELLED)

        # Mark expected attendance as absent
        Attendance.objects.filter(
            event=event,
            status=ATTENDANCE_EXPECTED,
        ).update(status=ATTENDANCE_ABSENT)

        log_action(
            actor=actor,
            action="cancel_event",
            entity_type="event",
            entity_id=event.id,
            after_data={"reason": reason},
        )

    return event

def apply_to_event(
    user,
    event: Event,
    post: EventPost = None,
    source: str = "browse",
    qr_token=None,
) -> EventApplication:
    """
    Volunteer applies to an event (or specific post in an event).
    Enforces active account, event status, deadlines, duplicates, and capacity gating.
    Lock order: Event -> Post -> Application.
    """
    ensure_active_account(user)

    if event.deleted_at is not None or event.status not in (EVENT_PUBLISHED, EVENT_ACTIVE):
        raise InvalidStateError(
            f"Event is not open for applications. Current status: '{event.status}'."
        )

    if event.application_deadline and event.application_deadline <= timezone.now():
        raise DomainError("Application deadline has passed.", code="deadline_passed")

    # Check existing active applications for (event, user)
    if EventApplication.objects.filter(
        event=event,
        user=user,
        status__in=ACTIVE_APPLICATION_STATUSES,
    ).exists():
        raise DomainError(
            "User already has an active application for this event.",
            code="duplicate_application",
        )

    if source == "qr_walkin":
        if not qr_token:
            raise DomainError("QR token required for walk-in registration.", code="qr_token_required")
        if qr_token.expires_at <= timezone.now() or qr_token.revoked_at is not None:
            raise DomainError("QR token is expired or revoked.", code="invalid_qr_token")

    conditions_snapshot = {
        "meals_provided": event.meals_provided,
        "lodging_provided": event.lodging_provided,
        "travel_support_provided": event.travel_support_provided,
        "compensation_kind": event.compensation_kind,
        "compensation_note": event.compensation_note,
        "onboarding_version": event.onboarding_version,
    }

    with transaction.atomic():
        # Lock event
        locked_event = Event.objects.select_for_update().get(id=event.id)
        locked_post = None
        if post:
            locked_post = EventPost.objects.select_for_update().get(id=post.id)

        from apps.ops.models import Assignment

        # Capacity check and status determination
        if locked_event.acceptance_mode == "auto":
            if locked_post:
                active_assignments_count = Assignment.objects.filter(
                    post=locked_post,
                    status__in=ACTIVE_ASSIGNMENT_STATUSES,
                ).count()
                capacity = locked_post.required_headcount - active_assignments_count
            else:
                if locked_event.max_participants:
                    accepted_count = EventApplication.objects.filter(
                        event=locked_event,
                        status=APPLICATION_ACCEPTED,
                    ).count()
                    capacity = locked_event.max_participants - accepted_count
                else:
                    capacity = 999999

            if capacity > 0:
                app_status = APPLICATION_ACCEPTED
                decided_at = timezone.now()
            else:
                app_status = APPLICATION_WAITLISTED
                decided_at = timezone.now()
        else:
            app_status = APPLICATION_PENDING
            decided_at = None

        application = EventApplication.objects.create(
            event=locked_event,
            post=locked_post,
            user=user,
            source=source,
            qr_token=qr_token,
            status=app_status,
            decided_at=decided_at,
            conditions_snapshot=conditions_snapshot,
        )

        if app_status == APPLICATION_ACCEPTED and locked_event.auto_assign_enabled:
            from apps.ops.services import create_assignment_from_application
            create_assignment_from_application(application)

        log_action(
            actor=user,
            action="apply_event",
            entity_type="event_application",
            entity_id=application.id,
            after_data={"status": app_status, "source": source},
        )

    return application


def accept_application(
    actor,
    application: EventApplication,
    post: EventPost = None,
):
    """
    Coordinator or Admin accepts an application.
    Verifies capacity on target post or event.
    Returns (application, assignment).
    """
    ensure_event_manager(actor, application.event)

    if application.status == APPLICATION_ACCEPTED:
        raise InvalidStateError("Application is already accepted.")

    if application.status not in (APPLICATION_PENDING, APPLICATION_WAITLISTED):
        raise InvalidStateError(
            f"Cannot accept application in status '{application.status}'."
        )

    target_post = post or application.post

    with transaction.atomic():
        locked_app = EventApplication.objects.select_for_update().get(id=application.id)
        from apps.ops.models import Assignment

        if target_post:
            locked_post = EventPost.objects.select_for_update().get(id=target_post.id)
            active_post_assigns = Assignment.objects.filter(
                post=locked_post,
                status__in=ACTIVE_ASSIGNMENT_STATUSES,
            ).count()
            if active_post_assigns >= locked_post.required_headcount:
                raise CapacityFullError("Target post capacity is full.")
            locked_app.post = locked_post
        elif locked_app.event.max_participants:
            accepted_apps_count = EventApplication.objects.filter(
                event=locked_app.event,
                status=APPLICATION_ACCEPTED,
            ).exclude(id=locked_app.id).count()
            if accepted_apps_count >= locked_app.event.max_participants:
                raise CapacityFullError("Event capacity is full.")

        locked_app.status = APPLICATION_ACCEPTED
        locked_app.decided_by = actor
        locked_app.decided_at = timezone.now()
        locked_app.save()

        from apps.ops.services import create_assignment_from_application
        assignment = create_assignment_from_application(locked_app, actor=actor)

        log_action(
            actor=actor,
            action="accept_application",
            entity_type="event_application",
            entity_id=locked_app.id,
        )

    return locked_app, assignment


def reject_application(
    actor,
    application: EventApplication,
    reason: str = "",
) -> EventApplication:
    """
    Coordinator or Admin rejects an application.
    """
    ensure_event_manager(actor, application.event)

    if application.status == APPLICATION_REJECTED:
        return application

    with transaction.atomic():
        application.status = APPLICATION_REJECTED
        application.decided_by = actor
        application.decided_at = timezone.now()
        if reason:
            application.note = reason
        application.save()

        log_action(
            actor=actor,
            action="reject_application",
            entity_type="event_application",
            entity_id=application.id,
            after_data={"reason": reason},
        )

    return application

def withdraw_application(
    user,
    application: EventApplication,
) -> EventApplication:
    """
    Volunteer withdraws own application.
    Cancels any active assignment.
    """
    if application.user_id != user.id:
        raise PermissionDeniedError("Cannot withdraw another volunteer's application.")

    if application.status not in (APPLICATION_PENDING, APPLICATION_ACCEPTED, APPLICATION_WAITLISTED):
        raise InvalidStateError(
            f"Cannot withdraw application in status '{application.status}'."
        )

    from apps.ops.models import Assignment

    with transaction.atomic():
        Assignment.objects.filter(
            application=application,
            status__in=ACTIVE_ASSIGNMENT_STATUSES,
        ).update(status=ASSIGNMENT_CANCELLED)

        application.status = APPLICATION_WITHDRAWN
        application.save()

        log_action(
            actor=user,
            action="withdraw_application",
            entity_type="event_application",
            entity_id=application.id,
        )

    return application

class PublicEventListView(APIView):
    permission_classes = [AllowAny]

    def get(self, request):
        qs = Event.objects.visible().filter(status__in=[EVENT_PUBLISHED, EVENT_ACTIVE])

        charity_field = request.query_params.get("charity_field")
        if charity_field:
            qs = qs.filter(charity_links__field_code_id=charity_field)

        date_from = request.query_params.get("date_from")
        if date_from:
            qs = qs.filter(start_at__gte=date_from)

        date_to = request.query_params.get("date_to")
        if date_to:
            qs = qs.filter(start_at__lte=date_to)

        q = request.query_params.get("q")
        if q:
            qs = qs.filter(
                Q(title__icontains=q)
                | Q(description__icontains=q)
                | Q(location_name__icontains=q)
                | Q(landmark_text__icontains=q)
            )

        qs = qs.distinct().order_by("start_at")
        return Response(EventSerializer(qs, many=True).data)

            .order_by("group_code", "id")
        )
        attendances = {
            att.user_id: att.status
            for att in Attendance.objects.filter(event=event)
        }

        roster = []
        for a in assignments:
            roster.append(
                {
                    "assignment_id": a.id,
                    "user_id": a.user_id,
                    "display_name": a.user.display_name,
                    "post_id": a.post_id,
                    "post_title": a.post.title if a.post else None,
                    "group_code": a.group_code,
                    "is_group_leader": a.is_group_leader,
                    "status": a.status,
                    "attendance_status": attendances.get(a.user_id, "none"),
                }
            )

        return Response({"roster": roster})


class EventUnassignedView(APIView):
    permission_classes = [IsAuthenticated, IsActiveAccount]

    def get(self, request, pk):
        event = get_object_or_404(Event.objects.visible(), pk=pk)
        ensure_event_manager(request.user, event)

        unassigned_apps = (
            EventApplication.objects.filter(event=event, status="accepted")
            .filter(~Q(assignments__status__in=ACTIVE_ASSIGNMENT_STATUSES))
            .distinct()
        )
        return Response(EventApplicationSerializer(unassigned_apps, many=True).data)


class ApplicationAcceptView(APIView):
    permission_classes = [IsAuthenticated, IsActiveAccount]

    def post(self, request, pk):
        app = get_object_or_404(EventApplication, pk=pk)
        post = None
        post_id = request.data.get("post_id")
        if post_id:
            post = get_object_or_404(app.event.posts, pk=post_id)

        accepted_app, assignment = accept_application(
            actor=request.user,
            application=app,
            post=post,
        )
        return Response(
            {
                "application": EventApplicationSerializer(accepted_app).data,
                "assigned": bool(assignment),
            }
        )
