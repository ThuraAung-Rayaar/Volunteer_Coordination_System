class AccountsApiTests(APITestCase):
    def setUp(self):
        self.phone = "+959987654321"

    def test_otp_flow_api_returns_token(self):
        # 1. Request OTP
        req_resp = self.client.post("/api/v1/auth/otp/request/", {"phone": self.phone})
        self.assertEqual(req_resp.status_code, status.HTTP_200_OK)
        self.assertEqual(req_resp.data.get("status"), "sent")
        code = req_resp.data.get("dev_code")
        self.assertIsNotNone(code)

        # 2. Verify OTP
        verify_resp = self.client.post("/api/v1/auth/otp/verify/", {"phone": self.phone, "code": code})
        self.assertEqual(verify_resp.status_code, status.HTTP_200_OK)
        self.assertIn("token", verify_resp.data)
        self.assertIn("user", verify_resp.data)
        self.assertTrue(verify_resp.data.get("created"))
        self.assertEqual(verify_resp.data["user"]["display_name"], f"Volunteer {self.phone[-4:]}")

    def test_me_patch_cannot_elevate_admin(self):
        # Create user & token
        user = User.objects.create(
            phone_hash="hash_user_patch",
            display_name="Regular Volunteer",
            is_admin=False,
        )
        token = issue_token_for_user(user)
        self.client.credentials(HTTP_AUTHORIZATION=f"Token {token.key}")

        # Attempt to patch regular fields - should succeed
        patch_resp = self.client.patch("/api/v1/me/", {"display_name": "Updated Name", "home_area": "Yangon"}, format="json")
        self.assertEqual(patch_resp.status_code, status.HTTP_200_OK)
        self.assertEqual(patch_resp.data["display_name"], "Updated Name")
        self.assertEqual(patch_resp.data["home_area"], "Yangon")

        # Attempt to escalate to admin - MUST fail with 400 Bad Request
        elevate_resp = self.client.patch("/api/v1/me/", {"is_admin": True}, format="json")
        self.assertEqual(elevate_resp.status_code, status.HTTP_400_BAD_REQUEST)
        self.assertEqual(elevate_resp.data["error"]["code"], "field_not_allowed")

        # Verify in DB that is_admin is still False
        user.refresh_from_db()
        self.assertFalse(user.is_admin)

from datetime import timedelta
from django.test import TestCase
from django.utils import timezone
from rest_framework.authtoken.models import Token

from apps.accounts.authentication import BlockedAwareTokenAuthentication
from apps.accounts.models import AuthOtp, User
from apps.accounts.services import request_otp, verify_otp
from apps.common.constants import ACCOUNT_ACTIVE, ACCOUNT_BLOCKED, ACCOUNT_CLOSED
from apps.common.exceptions import (
    AccountBlockedError,
    AccountClosedError,
    InvalidOtpError,
    TooManyAttemptsError,
)
from apps.common.security import compute_phone_hash, hash_otp


class AuthServicesTests(TestCase):
    def setUp(self):
        self.phone = "+959123456789"
        self.phone_hash = compute_phone_hash(self.phone)

    def test_otp_request_and_verify_creates_user(self):
        # Request OTP
        req_res = request_otp(self.phone, purpose="login")
        self.assertEqual(req_res.get("status"), "sent")
        code = req_res.get("dev_code")
        self.assertIsNotNone(code)

        # Verify OTP creates user and token
        user, token, created = verify_otp(self.phone, code, purpose="login")
        self.assertTrue(created)
        self.assertEqual(user.phone_hash, self.phone_hash)
        self.assertEqual(user.account_status, ACCOUNT_ACTIVE)
        self.assertIsNotNone(token)
        self.assertEqual(token.user, user)

        # Second verify for same user should not create a new user
        req_res2 = request_otp(self.phone, purpose="login")
        code2 = req_res2.get("dev_code")
        user2, token2, created2 = verify_otp(self.phone, code2, purpose="login")
        self.assertFalse(created2)
        self.assertEqual(user2.id, user.id)
        self.assertEqual(token2.key, token.key)

    def test_otp_reuse_fails(self):
        req_res = request_otp(self.phone, purpose="login")
        code = req_res.get("dev_code")

        # First verification succeeds
        verify_otp(self.phone, code, purpose="login")

        # Reusing the consumed OTP must raise InvalidOtpError
        with self.assertRaises(InvalidOtpError):
            verify_otp(self.phone, code, purpose="login")

    def test_otp_expired_fails(self):
        code = "123456"
        # Create an expired OTP directly
        AuthOtp.objects.create(
            phone_hash=self.phone_hash,
            code_hash=hash_otp(code),
            purpose="login",
            expires_at=timezone.now() - timedelta(seconds=10),
        )

        with self.assertRaises(InvalidOtpError):
            verify_otp(self.phone, code, purpose="login")

    def test_otp_max_attempts_locks(self):
        req_res = request_otp(self.phone, purpose="login")
        code = req_res.get("dev_code")
        wrong_code = "000000" if code != "000000" else "111111"

        # Fail 5 times with wrong code
        for _ in range(5):
            with self.assertRaises(InvalidOtpError):
                verify_otp(self.phone, wrong_code, purpose="login")

        # 6th attempt with correct code should fail with TooManyAttemptsError
        with self.assertRaises(TooManyAttemptsError):
            verify_otp(self.phone, code, purpose="login")

    def test_blocked_user_rejected_by_auth_class(self):
        # Create a blocked user and token
        user = User.objects.create(
            phone_hash=self.phone_hash,
            display_name="Blocked User",
            account_status=ACCOUNT_BLOCKED,
            blocked_until=timezone.now() + timedelta(days=1),
        )
        token = Token.objects.create(user=user)

        authenticator = BlockedAwareTokenAuthentication()

        with self.assertRaises(AccountBlockedError):
            authenticator.authenticate_credentials(token.key)

    def test_closed_account_cannot_request_otp(self):
        User.objects.create(
            phone_hash=self.phone_hash,
            display_name="Closed User",
            account_status=ACCOUNT_CLOSED,
        )
        with self.assertRaises(AccountClosedError):
            request_otp(self.phone, purpose="login")

class AccountsModelConstraintTests(SimpleTestCase):
    def test_blocked_until_constraint_violation(self):
        # 1. Active account with blocked_until MUST raise ValidationError
        invalid_user = User(
            phone_hash="hash_blocked_until_violation",
            phone_enc="enc_phone",
            display_name="Invalid Blocked User",
            account_status=ACCOUNT_ACTIVE,
            blocked_until=timezone.now() + timedelta(days=1),
        )
        with self.assertRaises(ValidationError):
            invalid_user.clean()

        # 2. Blocked account with blocked_until is valid
        valid_blocked_user = User(
            phone_hash="hash_valid_blocked",
            phone_enc="enc_phone",
            display_name="Valid Blocked User",
            account_status=ACCOUNT_BLOCKED,
            blocked_until=timezone.now() + timedelta(days=1),
        )
        # Should not raise validation error for blocked_until
        try:
            valid_blocked_user.clean()
        except ValidationError as e:
            self.assertNotIn("blocked_until", e.message_dict)

        # 3. Active account with blocked_until=None is valid
        valid_active_user = User(
            phone_hash="hash_valid_active",
            phone_enc="enc_phone",
            display_name="Valid Active User",
            account_status=ACCOUNT_ACTIVE,
            blocked_until=None,
        )
        try:
            valid_active_user.clean()
        except ValidationError as e:
            self.assertNotIn("blocked_until", e.message_dict)

        # Verify the CheckConstraint condition in Meta
        constraint = next(
            c for c in User._meta.constraints if c.name == "ck_users_blocked_until"
        )
        self.assertIsNotNone(constraint)
        self.assertEqual(
            constraint.condition,
            Q(blocked_until__isnull=True) | Q(account_status="blocked"),
        )

    def test_donation_consistency_constraint_violation(self):
        # 1. Willing to donate blood without is_willing_to_donate=True MUST fail
        invalid_blood_donor = User(
            phone_hash="hash_invalid_blood_donor",
            phone_enc="enc_phone",
            display_name="Invalid Blood Donor",
            is_willing_to_donate=False,
            is_willing_to_donate_blood=True,
            is_willing_to_donate_items=False,
        )
        with self.assertRaises(ValidationError):
            invalid_blood_donor.clean()

        # 2. Willing to donate items without is_willing_to_donate=True MUST fail
        invalid_items_donor = User(
            phone_hash="hash_invalid_items_donor",
            phone_enc="enc_phone",
            display_name="Invalid Items Donor",
            is_willing_to_donate=False,
            is_willing_to_donate_blood=False,
            is_willing_to_donate_items=True,
        )
        with self.assertRaises(ValidationError):
            invalid_items_donor.clean()

        # 3. Willing to donate=True with blood/items is valid
        valid_donor = User(
            phone_hash="hash_valid_donor",
            phone_enc="enc_phone",
            display_name="Valid Donor",
            is_willing_to_donate=True,
            is_willing_to_donate_blood=True,
            is_willing_to_donate_items=True,
        )
        try:
            valid_donor.clean()
        except ValidationError as e:
            self.assertNotIn("is_willing_to_donate", e.message_dict)

        # 4. Willing to donate=False with blood=False, items=False is valid
        valid_non_donor = User(
            phone_hash="hash_valid_non_donor",
            phone_enc="enc_phone",
            display_name="Valid Non Donor",
            is_willing_to_donate=False,
            is_willing_to_donate_blood=False,
            is_willing_to_donate_items=False,
        )
        try:
            valid_non_donor.clean()
        except ValidationError as e:
            self.assertNotIn("is_willing_to_donate", e.message_dict)

        # Verify the CheckConstraint condition in Meta
        constraint = next(
            c for c in User._meta.constraints if c.name == "ck_users_donation_consistency"
        )
        self.assertIsNotNone(constraint)
        self.assertEqual(
            constraint.condition,
            Q(is_willing_to_donate=True)
            | (Q(is_willing_to_donate_blood=False) & Q(is_willing_to_donate_items=False)),
        )

    def test_single_current_verification_image_constraint(self):
        # Verify UniqueConstraint specification on UserVerificationImage
        constraint = next(
            c for c in UserVerificationImage._meta.constraints
            if c.name == "uq_current_verification_image"
        )
        self.assertIsNotNone(constraint)
        self.assertEqual(list(constraint.fields), ["user", "image_type"])
        self.assertEqual(constraint.condition, Q(is_current=True))

        # Also verify review constraint on UserVerificationImage
        review_constraint = next(
            c for c in UserVerificationImage._meta.constraints
            if c.name == "ck_verification_review"
        )
        self.assertIsNotNone(review_constraint)
        self.assertEqual(
            review_constraint.condition,
            Q(status="pending") | (Q(reviewed_by__isnull=False) & Q(reviewed_at__isnull=False)),

class ProfileServicesTests(TestCase):

    def test_verification_image_transcoded_to_jpeg_and_old_deactivated(self):
        # 1. Upload first selfie (PNG format)
        file1 = create_dummy_uploaded_image("selfie1.png", format="PNG", color="blue")
        img1 = add_verification_image(self.user, IMAGE_MOCK_SELFIE, file1)

        self.assertTrue(img1.is_current)
        self.assertEqual(img1.file_format, "jpeg")
        self.assertTrue(img1.image_path.name.endswith(".jpg"))
        self.assertIsNotNone(img1.file_sha256)

        # User status should now be pending
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        file2 = create_dummy_uploaded_image("selfie2.png", format="PNG", color="green")

        # img1 must now be deactivated (is_current=False), img2 is current
        img1.refresh_from_db()
        self.assertFalse(img1.is_current)
        self.assertTrue(img2.is_current)
        self.assertNotEqual(img1.id, img2.id)

    def test_recompute_verification_status_logic(self):
        # Initially none
        self.assertEqual(self.user.verification_status, VERIFICATION_NONE)

        # Upload selfie -> pending
        file_selfie = create_dummy_uploaded_image("selfie.png")
        img_selfie = add_verification_image(self.user, IMAGE_MOCK_SELFIE, file_selfie)
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        # Upload ID -> pending
        file_id = create_dummy_uploaded_image("id.png")
        img_id = add_verification_image(self.user, IMAGE_ID_LICENSE, file_id)
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        # Approve selfie only -> still pending because ID is pending
        review_verification_image(self.admin, img_selfie, status="approved")
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_PENDING)

        # Approve ID -> both approved -> verified!
        review_verification_image(self.admin, img_id, status="approved")
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_VERIFIED)

        # If a new ID is uploaded and rejected -> status becomes rejected
        file_id2 = create_dummy_uploaded_image("id2.png")
        img_id2 = add_verification_image(self.user, IMAGE_ID_LICENSE, file_id2)
        review_verification_image(self.admin, img_id2, status="rejected", note="Blurry image")
        self.user.refresh_from_db()
        self.assertEqual(self.user.verification_status, VERIFICATION_REJECTED)

    def test_coordinator_approval_requires_verified_user(self):
        # Apply as coordinator
        profile = apply_as_coordinator(self.user, notes="Ready to lead events")
        self.assertEqual(profile.status, COORDINATOR_PENDING)

        # Non-admin cannot approve
        non_admin = User.objects.create(phone_hash="hash_regular", display_name="Regular")
        with self.assertRaises(PermissionDeniedError):
            approve_coordinator(non_admin, profile)

        # Admin cannot approve if user is unverified
        self.assertEqual(self.user.verification_status, VERIFICATION_NONE)
        with self.assertRaises(VerificationRequiredError):
            approve_coordinator(self.admin, profile)

        # Now verify user
        self.user.verification_status = VERIFICATION_VERIFIED
        self.user.save()

        # Approval succeeds
        approved_profile = approve_coordinator(self.admin, profile)
        self.assertEqual(approved_profile.status, COORDINATOR_APPROVED)
        self.assertEqual(approved_profile.approved_by, self.admin)

        # Suspend

        self.assertEqual(suspended.status, COORDINATOR_SUSPENDED)

        # Revoke
        revoked = revoke_coordinator(self.admin, suspended)
        self.assertEqual(revoked.status, COORDINATOR_REVOKED)

from django.test import SimpleTestCase, override_settings

from apps.common.security import (
    compute_phone_hash,

@override_settings(PHONE_HASH_HMAC_SECRET="test-hmac-secret-volunteer-system-2026")
class SecurityTests(SimpleTestCase):
    def test_phone_normalization_formats(self):
        # International with plus and spaces/dashes
        self.assertEqual(normalize_phone("+1 (555) 123-4567"), "+15551234567")
        self.assertEqual(normalize_phone("  +44 20 7946 0958  "), "+442079460958")
        self.assertEqual(normalize_phone("+95-9-123-456-789"), "+959123456789")

        # Local without plus
        self.assertEqual(normalize_phone("09-123-456-789"), "09123456789")
        self.assertEqual(normalize_phone(" (09) 1234 5678 "), "0912345678")

        # Empty / None handling
        self.assertEqual(normalize_phone(""), "")
        self.assertEqual(normalize_phone(None), "")

    def test_phone_hash_is_deterministic(self):
        # Same phone yields same hash
        h1 = compute_phone_hash("+959123456789")
        h2 = compute_phone_hash("+959123456789")
        self.assertEqual(h1, h2)
        self.assertEqual(len(h1), 64)  # HMAC-SHA256 hex digest length

        # Formatted phone yields identical hash due to normalization
        h3 = compute_phone_hash("+95-9-123-456-789")
        self.assertEqual(h1, h3)

        # Different phones yield different hashes
        h4 = compute_phone_hash("+959987654321")
        self.assertNotEqual(h1, h4)

    def test_encrypt_decrypt_phone_roundtrip(self):
        raw_phone = "+959123456789"
        encrypted = encrypt_phone(raw_phone)

        # Encrypted string must not be plain text
        self.assertNotEqual(encrypted, raw_phone)
        self.assertNotIn("959123456789", encrypted)

        # Decryption restores the normalized phone
        decrypted = decrypt_phone(encrypted)
        self.assertEqual(decrypted, raw_phone)

        # Test phone with spaces/dashes restores normalized form
        formatted_phone = "+1 (555) 234-5678"
        enc_formatted = encrypt_phone(formatted_phone)
        dec_formatted = decrypt_phone(enc_formatted)
        self.assertEqual(dec_formatted, "+15552345678")

    def test_raw_token_does_not_equal_hash(self):
        raw_token = generate_raw_token()
        self.assertTrue(len(raw_token) >= 32)

        hashed = hash_token(raw_token)
        self.assertEqual(len(hashed), 64)  # SHA-256 hex digest
        self.assertNotEqual(raw_token, hashed)

        # Hash is deterministic
        self.assertEqual(hash_token(raw_token), hashed)

    def test_hash_otp_deterministic(self):
        otp = generate_otp_code()
        self.assertEqual(len(otp), 6)
        self.assertTrue(otp.isdigit())

        h1 = hash_otp(otp)
        h2 = hash_otp(otp)
        self.assertEqual(h1, h2)
        self.assertEqual(len(h1), 64)
        self.assertNotEqual(otp, h1)

        # Different OTP codes yield different hashes
        diff_otp = "000000" if otp != "000000" else "111111"
        self.assertNotEqual(hash_otp(otp), hash_otp(diff_otp))

        coordinator = self._create_dummy_user(1)

        invalid_event = Event(
            coordinator=coordinator,

            start_at=now + timedelta(days=2),
            end_at=now + timedelta(days=1),
        )
        with self.assertRaises(ValidationError):
            invalid_event.clean()

        # 2. end_at equal to start_at MUST raise ValidationError
        invalid_equal_event = Event(
            coordinator=coordinator,
            title="Equal Time Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=1),
        )
        with self.assertRaises(ValidationError):
            invalid_equal_event.clean()

        # 3. end_at > start_at is valid
        valid_event = Event(
            coordinator=coordinator,
            title="Valid Time Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
        )
        try:
            valid_event.clean()
        except ValidationError as e:
            self.assertNotIn("end_at", e.message_dict)

        # 4. application_deadline > start_at MUST raise ValidationError
        invalid_deadline_event = Event(
            coordinator=coordinator,
            title="Late Deadline Event",
            start_at=now + timedelta(days=1),
            end_at=now + timedelta(days=2),
            application_deadline=now + timedelta(days=3),
        )
        with self.assertRaises(ValidationError):
            invalid_deadline_event.clean()

        # 5. Verify CheckConstraint definitions in Meta
        time_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_time")
        self.assertEqual(time_constraint.condition, Q(end_at__gt=F("start_at")))

        deadline_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_deadline")
        self.assertEqual(
            deadline_constraint.condition,
            Q(application_deadline__isnull=True) | Q(application_deadline__lte=F("start_at")),
        )

class OpsModelConstraintTests(SimpleTestCase):
    def _create_dummy_user(self, user_id=1):
        return User(id=user_id, phone_hash=f"hash_{user_id}", display_name=f"User {user_id}")

    def _create_dummy_event(self, event_id=1):
        return Event(id=event_id, title=f"Event {event_id}")

    def test_qr_max_uses_boundary(self):
        event = self._create_dummy_event(1)
        user = self._create_dummy_user(1)

        # 1. max_uses <= 0 MUST raise ValidationError
        invalid_max_uses = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_invalid_max",
            max_uses=0,
            used_count=0,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        with self.assertRaises(ValidationError):
            invalid_max_uses.clean()

        # 2. used_count < 0 MUST raise ValidationError
        invalid_negative_used = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_negative_used",
            max_uses=10,
            used_count=-1,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        with self.assertRaises(ValidationError):
            invalid_negative_used.clean()

        # 3. used_count > max_uses MUST raise ValidationError
        invalid_over_used = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_over_used",
            max_uses=5,
            used_count=6,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        with self.assertRaises(ValidationError):
            invalid_over_used.clean()

        # 4. used_count <= max_uses is valid
        valid_qr = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_valid_qr",
            max_uses=5,
            used_count=3,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        try:
            valid_qr.clean()
        except ValidationError as e:
            self.assertNotIn("used_count", str(e))
            self.assertNotIn("max_uses", str(e))

        # 5. Verify CheckConstraints in Meta
        c_max = next(c for c in QrToken._meta.constraints if c.name == "ck_qr_max_uses")
        self.assertEqual(c_max.condition, Q(max_uses__isnull=True) | Q(max_uses__gt=0))

        c_nonneg = next(c for c in QrToken._meta.constraints if c.name == "ck_qr_used_count_nonnegative")
        self.assertEqual(c_nonneg.condition, Q(used_count__gte=0))

        c_limit = next(c for c in QrToken._meta.constraints if c.name == "ck_qr_used_count")
        self.assertEqual(c_limit.condition, Q(max_uses__isnull=True) | Q(used_count__lte=F("max_uses")))

    def test_assignment_unique_active_per_event_and_application(self):
        # 1. Verify UniqueConstraints
        uq_active = next(c for c in Assignment._meta.constraints if c.name == "uq_assignment_active")
        self.assertEqual(list(uq_active.fields), ["event", "user"])
        self.assertEqual(uq_active.condition, ~Q(status="cancelled"))

        uq_app = next(c for c in Assignment._meta.constraints if c.name == "uq_assignment_application")
        self.assertEqual(list(uq_app.fields), ["application"])
        self.assertEqual(uq_app.condition, ~Q(status="cancelled"))

        # 2. Actor check constraint: auto assignment with assigned_by_user MUST raise ValidationError
        event = self._create_dummy_event(1)
        user = self._create_dummy_user(1)
        app = EventApplication(id=1, event=event, user=user)

        invalid_auto_actor = Assignment(
            event=event,
            application=app,
            user=user,
            assigned_by="auto",
            assigned_by_user=user,
        )
        with self.assertRaises(ValidationError):
            invalid_auto_actor.clean()

        # 3. Manual assignment without assigned_by_user MUST raise ValidationError
        invalid_manual_actor = Assignment(
            event=event,
            application=app,
            user=user,
            assigned_by="coordinator",
            assigned_by_user=None,
        )
        with self.assertRaises(ValidationError):

    def test_otp_request_and_verify_creates_user(self):

        req_res = request_otp(self.phone, purpose="login")
        self.assertEqual(req_res.get("status"), "sent")
        code = req_res.get("dev_code")
        self.assertIsNotNone(code)

        # Verify OTP creates user and token
        user, token, created = verify_otp(self.phone, code, purpose="login")
        self.assertTrue(created)
        self.assertEqual(user.phone_hash, self.phone_hash)
        self.assertEqual(user.account_status, ACCOUNT_ACTIVE)
        self.assertIsNotNone(token)
        self.assertEqual(token.user, user)

        # Second verify for same user should not create a new user
        req_res2 = request_otp(self.phone, purpose="login")
        code2 = req_res2.get("dev_code")
        user2, token2, created2 = verify_otp(self.phone, code2, purpose="login")
        self.assertFalse(created2)
        self.assertEqual(user2.id, user.id)
        self.assertEqual(token2.key, token.key)

        req_res = request_otp(self.phone, purpose="login")
        code = req_res.get("dev_code")

        # First verification succeeds
        verify_otp(self.phone, code, purpose="login")

        # Reusing the consumed OTP must raise InvalidOtpError
        with self.assertRaises(InvalidOtpError):
            verify_otp(self.phone, code, purpose="login")

    def test_otp_expired_fails(self):
        code = "123456"
        # Create an expired OTP directly
        AuthOtp.objects.create(
            phone_hash=self.phone_hash,
            code_hash=hash_otp(code),
            purpose="login",
            expires_at=timezone.now() - timedelta(seconds=10),
        )

        with self.assertRaises(InvalidOtpError):
            verify_otp(self.phone, code, purpose="login")

    def test_otp_max_attempts_locks(self):
        req_res = request_otp(self.phone, purpose="login")
        code = req_res.get("dev_code")
        wrong_code = "000000" if code != "000000" else "111111"

        # Fail 5 times with wrong code
        for _ in range(5):

def create_dummy_uploaded_image(filename="sample.png", format="PNG", color="red"):
    buffer = BytesIO()

@override_settings(PHONE_HASH_HMAC_SECRET="test-hmac-secret-volunteer-system-2026")
class SecurityTests(SimpleTestCase):
    def test_phone_normalization_formats(self):
        # International with plus and spaces/dashes
        self.assertEqual(normalize_phone("+1 (555) 123-4567"), "+15551234567")
        self.assertEqual(normalize_phone("  +44 20 7946 0958  "), "+442079460958")
        self.assertEqual(normalize_phone("+95-9-123-456-789"), "+959123456789")

        # Local without plus
        self.assertEqual(normalize_phone("09-123-456-789"), "09123456789")
        self.assertEqual(normalize_phone(" (09) 1234 5678 "), "0912345678")

        # Empty / None handling
        self.assertEqual(normalize_phone(""), "")
        self.assertEqual(normalize_phone(None), "")

    def test_phone_hash_is_deterministic(self):
        # Same phone yields same hash
        h1 = compute_phone_hash("+959123456789")
        h2 = compute_phone_hash("+959123456789")
        self.assertEqual(h1, h2)
        self.assertEqual(len(h1), 64)  # HMAC-SHA256 hex digest length

        # Formatted phone yields identical hash due to normalization
        h3 = compute_phone_hash("+95-9-123-456-789")
        self.assertEqual(h1, h3)

        # Different phones yield different hashes
        h4 = compute_phone_hash("+959987654321")
        self.assertNotEqual(h1, h4)

    def test_encrypt_decrypt_phone_roundtrip(self):
        raw_phone = "+959123456789"
        encrypted = encrypt_phone(raw_phone)

        # Encrypted string must not be plain text
        self.assertNotEqual(encrypted, raw_phone)
        self.assertNotIn("959123456789", encrypted)

        # Decryption restores the normalized phone
        decrypted = decrypt_phone(encrypted)
        self.assertEqual(decrypted, raw_phone)

        # Test phone with spaces/dashes restores normalized form
        formatted_phone = "+1 (555) 234-5678"
        enc_formatted = encrypt_phone(formatted_phone)
        dec_formatted = decrypt_phone(enc_formatted)
        self.assertEqual(dec_formatted, "+15552345678")

    def test_raw_token_does_not_equal_hash(self):

    def test_event_time_constraint(self):
        coordinator = self._create_dummy_user(1)

        # 1. end_at earlier than start_at MUST raise ValidationError
        invalid_event = Event(

            invalid_deadline_event.clean()

        # 5. Verify CheckConstraint definitions in Meta
        time_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_time")
        self.assertEqual(time_constraint.condition, Q(end_at__gt=F("start_at")))

        deadline_constraint = next(c for c in Event._meta.constraints if c.name == "ck_events_deadline")
        self.assertEqual(
            deadline_constraint.condition,
            Q(application_deadline__isnull=True) | Q(application_deadline__lte=F("start_at")),
        )

    def test_qr_max_uses_boundary(self):
        event = self._create_dummy_event(1)
        user = self._create_dummy_user(1)

        # 1. max_uses <= 0 MUST raise ValidationError
        invalid_max_uses = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_invalid_max",
            max_uses=0,
            used_count=0,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        with self.assertRaises(ValidationError):
            invalid_max_uses.clean()

        # 2. used_count < 0 MUST raise ValidationError
        invalid_negative_used = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_negative_used",
            max_uses=10,
            used_count=-1,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        with self.assertRaises(ValidationError):
            invalid_negative_used.clean()

        # 3. used_count > max_uses MUST raise ValidationError
        invalid_over_used = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_over_used",
            max_uses=5,
            used_count=6,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        with self.assertRaises(ValidationError):
            invalid_over_used.clean()

        # 4. used_count <= max_uses is valid
        valid_qr = QrToken(
            event=event,
            created_by=user,
            purpose=QR_CHECKIN,
            token_hash="hash_valid_qr",
            max_uses=5,
            used_count=3,
            expires_at=timezone.now() + timedelta(hours=1),
        )
        try:
            valid_qr.clean()
        except ValidationError as e:
            self.assertNotIn("used_count", str(e))
            self.assertNotIn("max_uses", str(e))

        # 5. Verify CheckConstraints in Meta
        c_max = next(c for c in QrToken._meta.constraints if c.name == "ck_qr_max_uses")
        self.assertEqual(c_max.condition, Q(max_uses__isnull=True) | Q(max_uses__gt=0))

        c_nonneg = next(c for c in QrToken._meta.constraints if c.name == "ck_qr_used_count_nonnegative")
        self.assertEqual(c_nonneg.condition, Q(used_count__gte=0))

        c_limit = next(c for c in QrToken._meta.constraints if c.name == "ck_qr_used_count")
        self.assertEqual(c_limit.condition, Q(max_uses__isnull=True) | Q(used_count__lte=F("max_uses")))

    def test_qr_generation_returns_raw_token_and_stores_hash(self):
        qr_token, raw_token = generate_qr_token(
            actor=self.coordinator,
            event=self.event,
            purpose=QR_WALKIN_REGISTRATION,
            max_uses=5,
            ttl_minutes=120,
        )

        self.assertIsNotNone(raw_token)
        self.assertNotEqual(qr_token.token_hash, raw_token)
        self.assertEqual(qr_token.token_hash, hash_token(raw_token))
        self.assertEqual(qr_token.used_count, 0)
        self.assertEqual(qr_token.max_uses, 5)

        # Non-manager cannot generate token
        with self.assertRaises(PermissionDeniedError):
            generate_qr_token(self.volunteer, self.event, QR_WALKIN_REGISTRATION)

    def test_resolve_does_not_increment_used_count(self):
        qr_token, raw_token = generate_qr_token(
            actor=self.coordinator,
            event=self.event,
            purpose=QR_WALKIN_REGISTRATION,
            max_uses=3,
        )

        preview = resolve_qr_token(raw_token)
        self.assertEqual(preview["event_id"], self.event.id)
        self.assertEqual(preview["title"], self.event.title)
        self.assertEqual(preview["purpose"], QR_WALKIN_REGISTRATION)
        self.assertEqual(preview["used_count"], 0)

        # used_count in DB must remain 0
        qr_token.refresh_from_db()
        self.assertEqual(qr_token.used_count, 0)

    def test_redeem_increments_used_count_and_enforces_max_uses(self):
        qr_token, raw_token = generate_qr_token(
            actor=self.coordinator,
            event=self.event,
            purpose=QR_WALKIN_REGISTRATION,
            max_uses=1,

        self.assertEqual(qr_token.used_count, 1)

        # 2. Second redemption exceeds max_uses -> CapacityFullError
        with self.assertRaises(CapacityFullError):
            redeem_qr_token(self.volunteer2, raw_token)

        with self.assertRaises(DomainError) as ctx_exp:
            resolve_qr_token(raw_expired)
        self.assertEqual(ctx_exp.exception.code, "expired_token")

        with self.assertRaises(DomainError):
            redeem_qr_token(self.volunteer, raw_expired)

        # 2. Revoked token
        qr_revoked, raw_revoked = generate_qr_token(
            self.coordinator,
            self.event,
            QR_WALKIN_REGISTRATION,
